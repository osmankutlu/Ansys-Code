def _style_button(b, kind):
    b.FlatStyle = FlatStyle.Flat
    if kind == u"primary":
        b.BackColor = THEME_BLUE
        b.ForeColor = Color.White
        b.Font      = Font(b.Font, FontStyle.Bold)
    elif kind == u"danger":
        b.BackColor = THEME_STOP
        b.ForeColor = Color.White
        b.Font      = Font(b.Font, FontStyle.Bold)
    else:
        b.BackColor = THEME_GREY
        b.ForeColor = Color.Black

def apply_theme(root):
    for c in root.Controls:
        try:
            if isinstance(c, Button):
                t = c.Text
                if t == u"Stop":
                    _style_button(c, u"danger")
                elif t in PRIMARY_TEXTS:
                    _style_button(c, u"primary")
                else:
                    _style_button(c, u"secondary")
            elif isinstance(c, GroupBox):
                saved = [(k, k.ForeColor) for k in c.Controls]
                c.ForeColor = THEME_BLUE_DARK
                for (k, fc) in saved:
                    k.ForeColor = fc
                apply_theme(c)
            elif isinstance(c, Panel):
                apply_theme(c)
        except:
            pass

def format_duration(elapsed_sec):
    if elapsed_sec < 60:
        return str(elapsed_sec) + u" sec"
    else:
        mins = elapsed_sec // 60
        secs = elapsed_sec - mins * 60
        return str(mins) + u" min " + str(secs) + u" sec"

def disp_type(cmd_type):
    if cmd_type == u"friction":
        return u"FRICTION"
    return u"PRELOAD"

def fmt_val(x):
    t = (u"%.8f" % x).rstrip(u"0").rstrip(u".")
    if t == u"" or t == u"-":
        t = u"0"
    return t

def get_existing_cid(contact):
    for child in contact.Children:
        if "CommandSnippet" not in child.GetType().ToString():
            continue
        try:
            matches = re.findall(r"cid_(\d+)\s*=\s*cid", child.Input)
            if matches:
                return int(matches[0])
        except:
            pass
    return None

# ---------------------------------------------------------------
# Tum kontaklardaki max cid numarasini bul
# ---------------------------------------------------------------
def find_max_existing_cid(connections):
    max_cid = 0
    for conn_group in connections.Children:
        for contact in conn_group.Children:
            if "ContactRegion" not in contact.GetType().ToString():
                continue
            cid = get_existing_cid(contact)
            if cid is not None and cid > max_cid:
                max_cid = cid
    return max_cid

def build_contact_tree(connections):
    tree = []
    for conn_group in connections.Children:
        group_contacts = []
        for contact in conn_group.Children:
            if "ContactRegion" in contact.GetType().ToString():
                group_contacts.append(contact)
        if group_contacts:
            tree.append((conn_group.Name, group_contacts))
    return tree

def make_cmd_line(cmd_type, cid_n, cmd_value):
    if cmd_type == u"friction":
        return u"MP,MU,cid_" + str(cid_n) + u"," + cmd_value
    else:
        return u"Rmodif,cid_" + str(cid_n) + u",10," + cmd_value

def cmd_prefix(cmd_type):
    if cmd_type == u"friction":
        return u"MP,MU,"
    else:
        return u"Rmodif,"

def cmd_line_key(cmd_type, cid_n):
    # Belirli bir cid satirini tanimak icin (cid_1 ile cid_10 karismasin diye sondaki virgul var)
    if cmd_type == u"friction":
        return (u"MP,MU,cid_" + str(cid_n) + u",").upper()
    else:
        return (u"Rmodif,cid_" + str(cid_n) + u",10,").upper()

# ---------------------------------------------------------------
# Mevcut ayni tip+step command'i bul
# ---------------------------------------------------------------
def find_existing_solution_cmd(analysis, cmd_type, step_num):
    prefix = cmd_prefix(cmd_type)
    for child in analysis.Children:
        if "CommandSnippet" not in child.GetType().ToString():
            continue
        try:
            if child.Name.startswith(u"PRELOAD_ITER_STEP"):
                continue
        except:
            pass
        try:
            snum = None
            try:
                snum = int(child.StepNumber)
            except:
                pass
            try:
                snum = int(child.InternalObject.StepNumber)
            except:
                pass
            if snum != step_num:
                continue
            if prefix.upper() in child.Input.upper():
                return child
        except:
            pass
    return None

# ---------------------------------------------------------------
# Mevcut command'e ekle veya yeni olustur
# ---------------------------------------------------------------
def add_or_update_solution_cmd(analysis, cmd_type, cmd_value, cid_list, step_num):
    existing  = find_existing_solution_cmd(analysis, cmd_type, step_num)
    new_lines = [make_cmd_line(cmd_type, cid_n, cmd_value) for cid_n in cid_list]

    if existing is not None:
        current_lines = [l for l in existing.Input.split(u"\n") if l.strip()]
        for line in new_lines:
            if line not in current_lines:
                current_lines.append(line)
        existing.Input = u"\n".join(current_lines)
        return existing, False
    else:
        cmd       = analysis.AddCommandSnippet()
        cmd.Name  = u"APDL_" + cmd_type.upper() + u"_STEP" + str(step_num)
        cmd.Input = u"\n".join(new_lines)
        cmd.StepSelectionMode = SequenceSelectionType.ByNumber
        cmd.StepNumber        = step_num
        return cmd, True

# ---------------------------------------------------------------
# Iterasyon icin: belirli cid satirinin DEGERINI degistir
# (satir varsa yerinde guncelle, yoksa ekle, command yoksa olustur)
# ---------------------------------------------------------------
def set_cmd_value(analysis, cmd_type, step_num, cid_n, value_str):
    existing = find_existing_solution_cmd(analysis, cmd_type, step_num)
    if existing is None:
        add_or_update_solution_cmd(analysis, cmd_type, value_str, [cid_n], step_num)
        return
    key       = cmd_line_key(cmd_type, cid_n)
    new_line  = make_cmd_line(cmd_type, cid_n, value_str)
    lines     = [l.strip() for l in existing.Input.split(u"\n") if l.strip()]
    found     = False
    for i in range(len(lines)):
        if lines[i].upper().startswith(key):
            lines[i] = new_line
            found = True
    if not found:
        lines.append(new_line)
    existing.Input = u"\n".join(lines)

# ---------------------------------------------------------------
# CID atama (hem manuel hem iterasyon modu kullanir)
# ---------------------------------------------------------------
def assign_cids(selected, start_cid, prog, cancelled):
    next_cid = start_cid
    cid_list = []
    reused   = 0
    created  = 0
    total    = len(selected)
    for idx, (contact, existing_cid) in enumerate(selected):
        if cancelled[0]:
            break
        prog.update(u"Adding CID commands...  (" + str(idx + 1) + u"/" + str(total) + u")",
                    contact.Name)
        if existing_cid is not None:
            cid_list.append(existing_cid)
            reused += 1
        else:
            next_cid += 1
            cid_list.append(next_cid)
            cmd_c       = contact.AddCommandSnippet()
            cmd_c.Name  = u"Contact_Command_" + str(next_cid)
            cmd_c.Input = u"cid_" + str(next_cid) + u" = cid"
            created += 1
    return cid_list, next_cid, reused, created

def within_tol(F, target, tol_pct):
    return abs(F - target) <= abs(target) * tol_pct / 100.0

def next_value(hist, target):
    # hist: [(x, F), ...]   son nokta = hist[-1]
    x1, f1 = hist[-1]
    prev = None
    for k in range(len(hist) - 2, -1, -1):
        if abs(hist[k][0] - x1) > 1e-12 * max(1.0, abs(x1)):
            prev = hist[k]
            break
    xn = None
    if prev is not None and abs(f1 - prev[1]) > 1e-12:
        # sekant: interpolasyon / ekstrapolasyon
        xn = x1 + (target - f1) * (x1 - prev[0]) / (f1 - prev[1])
    else:
        # oran-orantı
        if abs(f1) < 1e-9:
            xn = x1 * 2.0
        else:
            xn = x1 * target / f1
    if xn <= 0:
        xn = x1 * 0.5
    if xn > 5.0 * x1:
        xn = 5.0 * x1
    if xn < 0.1 * x1:
        xn = 0.1 * x1
    return xn

def rs_norm(n):
    return n.replace(" ", "").lower()

def rs_set_enum(obj, prop, candidates):
    """Enum ozelligini isimle atar; zaten ayni ise dokunmaz."""
    try:
        cur = getattr(obj, prop)
    except Exception:
        return False
    for c in candidates:
        if rs_norm(unicode(str(cur))) == rs_norm(c):
            return True
    try:
        t = cur.GetType()
        names = list(System.Enum.GetNames(t))
    except Exception:
        return False
    norm = dict([(rs_norm(n), n) for n in names])
    for c in candidates:
        if rs_norm(c) in norm:
            try:
                setattr(obj, prop, System.Enum.Parse(t, norm[rs_norm(c)]))
                return True
            except Exception:
                return False
    return False

def rs_parse_solve_out(workdir):
    info = {"restart_from": None, "elapsed": None, "cp": None}
    try:
        f = open(os.path.join(workdir, u"solve.out"), "r")
        try:
            lines = f.read().splitlines()
        finally:
            f.close()
    except Exception:
        return info
    for l in lines:
        m = re.search(r"RESTART PREVIOUSLY RUN ANALYSIS FROM LOADSTEP\s+(\d+)", l)
        if m:
            info["restart_from"] = int(m.group(1))
        m = re.search(r"Elapsed Time \(sec\)\s*=\s*([0-9.]+)", l)
        if m:
            info["elapsed"] = float(m.group(1))
        m = re.search(r"CP Time\s+\(sec\)\s*=\s*([0-9.]+)", l)
        if m:
            info["cp"] = float(m.group(1))
    return info

        try:
            f = open(os.path.join(workdir, u"solve.out"), "r")

                n = 0
                for line in f:

                    if n > 600:
                        break
                    if "_WB_PROJECTSCRATCH_DIR" in line:
                        m = re.search(r"=\s*['\"]?([^'\"\r\n]+)", line)
                        if m:
                            d = m.group(1).strip()
                            if os.path.isdir(d) and d not in dirs:
                                dirs.append(d)
                        break
            finally:
                f.close()
        except Exception:
            pass
    return dirs

def rs_send_abort(workdir):
    for d in rs_abort_dirs(workdir):
        try:
            f = open(os.path.join(d, u"file.abt"), "w")
            f.write("nonlinear\n")
            f.close()
        except Exception:
            pass

def rs_clear_abort(workdir):
    for d in rs_abort_dirs(workdir):
        try:
            pth = os.path.join(d, u"file.abt")
            if os.path.exists(pth):
                os.remove(pth)
        except Exception:
            pass

def solve_interruptible(analysis, cancelled, workdir):
    """
    Cozumu ASENKRON baslatir (Solve(False)) ve beklerken arayuzu isler; boylece Stop
    cozum sirasinda da calisir: Stop'a basilirsa solver'a abort dosyasi (file.abt) yazilir.
    Donen: (hata metni veya None, iptal edildi mi)
    Solution.ObjectState okunamiyorsa eski davranisa (bekleyen Solve(True)) doner.
    """
    if not rs_state_readable(analysis):
        try:
            analysis.Solve(True)
            return None, False
        except Exception as ex:
            return unicode(str(ex)), False
    t0 = time.time()
    try:
        analysis.Solve(False)
    except Exception as ex:
        return unicode(str(ex)), False
    seen_busy  = False
    abort_sent = False
    abort_t    = 0.0
    while True:
        Application.DoEvents()
        Thread.Sleep(200)
        busy = rs_is_solving(analysis)
        if busy:
            seen_busy = True
        if cancelled[0] and not abort_sent:
            rs_send_abort(workdir)
            abort_sent = True
            abort_t = time.time()
        if seen_busy and not busy:
            break
        if (not seen_busy) and (time.time() - t0 > 60):
            if abort_sent:
                break
            return u"Could not detect that the solver started (the solution never reported 'Solving').", False
        if abort_sent and (time.time() - abort_t > 120):
            break
    if abort_sent:
        rs_clear_abort(workdir)
    return None, bool(cancelled[0])

class RestartHelper(object):
    """
        rh = RestartHelper(analysis, workdir);  rh.setup()       # ayarlari kaydet + restart ayarlarini yap
        rh.scratch();  info = rh.solve()                         # sifirdan (restart noktalari uretilir)
        if rh.from_end_of_step(k):                               # k. step sonundan restart
            lines, warn = rh.carry_lines(k)                      # restart'ta kaybolan komutlar
            carry = rs_add_carry(analysis, k + 1, lines)
        info = rh.solve()                                        # info["restart_from"] == k ise restart kullanildi
        rh.restore()                                             # kullanicinin ayarlarini oldugu gibi geri yukle
    """
    PROPS = ["GenerateRestartPoints", "SaveAtLoadStep", "SaveAtSubstep", "RetainFilesAfterFullSolve",
             "CombineRestartFiles", "DeleteUnneededFiles", "RestartType", "CurrentRestartPoint"]
    RE_RMODIF = re.compile(r"^\s*Rmodif\s*,\s*([^,\s]+)\s*,\s*([^,\s]+)\s*,", re.I)
    RE_MP     = re.compile(r"^\s*MP\s*,\s*MU\s*,\s*([^,\s]+)\s*,", re.I)

    def __init__(self, analysis, workdir, combine=u"Yes", mid=False):
        self.analysis = analysis
        self.settings = analysis.AnalysisSettings
        self.workdir  = workdir
        self.combine  = combine
        self.mid      = mid          # True: her substep'te restart noktasi uret (step ortasindan restart icin)
        self.saved    = {}

    def setup(self):
        s = self.settings
        for p in RestartHelper.PROPS:
            try:
                self.saved[p] = getattr(s, p)
            except Exception:
                pass
        rs_set_enum(s, "GenerateRestartPoints", ["Manual"])
        rs_set_enum(s, "SaveAtLoadStep", ["All"])
        rs_set_enum(s, "SaveAtSubstep", ["All"])
        rs_set_enum(s, "RetainFilesAfterFullSolve", ["Yes"])
        rs_set_enum(s, "CombineRestartFiles", [self.combine])
        try:
            if getattr(s, "DeleteUnneededFiles") is not False:
                s.DeleteUnneededFiles = False
        except Exception:
            pass
        rs_set_enum(s, "RestartType", ["Manual"])

    def restore(self):
        """Kullanicinin restart ayarlarini bulundugu gibi geri yukler."""
        s = self.settings
        for p in ["RestartType", "CurrentRestartPoint", "GenerateRestartPoints", "SaveAtLoadStep",
                  "SaveAtSubstep", "RetainFilesAfterFullSolve", "CombineRestartFiles", "DeleteUnneededFiles"]:
            if p in self.saved:
                try:
                    setattr(s, p, self.saved[p])
                except Exception:
                    pass

    def scratch(self):
        rs_set_enum(self.settings, "RestartType", ["Manual"])
        try:
            self.settings.CurrentRestartPoint = 0
        except Exception:
            pass

    def find_point(self, loadstep, end_time=None):
        s = self.settings
        rs_set_enum(s, "RestartType", ["Manual"])
        limit = 20
        try:
            limit = max(limit, int(s.NumberOfRestartPoints) + 3)
        except Exception:
            pass
        for idx in range(0, limit):
            try:
                s.CurrentRestartPoint = idx
            except Exception:
                break
            try:
                ls = int(s.RestartAtLoadStep)
                tm = float(s.RestartAtTime)
            except Exception:
                continue
            if ls == loadstep and (end_time is None or abs(tm - end_time) <= 1e-6 * max(1.0, abs(end_time))):
                return idx
        return None

    def from_end_of_step(self, k):
        """k. load step'in sonundaki restart noktasini secer; yoksa sifirdan baslar (False)."""
        if k is None or k < 1:
            self.scratch()
            return False
        try:
            et = rs_step_end_time(self.settings, k)
        except Exception:
            et = None
        idx = self.find_point(k, et)
        if idx is None:
            self.scratch()
            return False
        try:
            self.settings.CurrentRestartPoint = idx
        except Exception:
            self.scratch()
            return False
        return True

    def save_last_only(self):
        """Sonraki cozumlerde sadece son substep noktalari uretilsin (restart'a ihtiyac kalmadiginda)."""
        rs_set_enum(self.settings, "SaveAtSubstep", ["Last"])

    def _pt(self, idx):
        s = self.settings
        try:
            s.CurrentRestartPoint = idx
        except Exception:
            return None
        try:
            return (int(s.RestartAtLoadStep), int(s.RestartAtSubstep), float(s.RestartAtTime))
        except Exception:
            return None

    def point_at_time(self, target_time):
        """
        Zamani target_time olan restart noktasinin indexi (yoksa None). Indexler zamanla artar,
        liste sonunda degerler tekrar eder; bu yuzden ikili aramayla az sayida deneme yapilir.
        """
        eps = 1e-6 * max(1.0, abs(target_time))
        hi = 1
        while True:
            pt = self._pt(hi)
            if pt is None:
                return None
            if pt[2] >= target_time - eps:
                break
            if hi >= 8192:
                return None
            hi *= 2
        lo = 0
        while lo < hi:
            mid = (lo + hi) // 2
            pt = self._pt(mid)
            if pt is None:
                return None
            if pt[2] >= target_time - eps:
                hi = mid
            else:
                lo = mid + 1
        pt = self._pt(lo)
        if pt is None or abs(pt[2] - target_time) > eps:
            return None
        return lo

    def last_point_of_step(self, p):
        """Listede load step'i p olan SON noktanin indexi (zaman eslesmesi bulunamazsa yedek yontem)."""
        best = None
        prev = None
        same = 0
        for idx in range(0, 4000):
            pt = self._pt(idx)
            if pt is None:
                break
            if pt == prev:
                same += 1
                if same >= 2:
                    break
                continue
            same = 0
            prev = pt
            if pt[0] > p:
                break
            if pt[0] == p:
                best = idx
        return best

    def choose(self, first_step, mid):
        """
        first_step: degisen en erken load step. Uygun restart noktasini secer (CurrentRestartPoint ayarlanir).
        mid=True : bir onceki step p'nin SON substep'inden bir oncekinden basla; p tek substep ise p'nin basindan
        mid=False: p'nin sonundan basla
        Donen: sozluk (idx, mode, expect, carry_step, carry_upto, skipped) veya None (sifirdan coz)
            expect     : solve.out'ta beklenen restart load step'i
            carry_step : carry snippet'inin ekleneceği step;  carry_upto: bu step'e kadarki komutlar yeniden uygulanir
        """
        p = first_step - 1
        if p < 1:
            return None
        try:
            et = rs_step_end_time(self.settings, p)
        except Exception:
            return None
        rs_set_enum(self.settings, "RestartType", ["Manual"])
        idx_end = self.point_at_time(et)
        if idx_end is None:
            idx_end = self.last_point_of_step(p)       # listede 'Load Step p' ile baslayan son nokta
        if idx_end is None:
            return None
        pt_end = self._pt(idx_end)
        if pt_end is None or pt_end[0] != p:
            return None
        plan = {"idx": idx_end, "mode": "end", "expect": p, "carry_step": first_step,
                "carry_upto": p, "skipped": p}
        if mid and idx_end >= 1:
            pt_prev = self._pt(idx_end - 1)
            if pt_prev is not None:
                if pt_prev[0] == p and pt_prev[1] >= 1 and pt_prev[1] < pt_end[1]:
                    plan = {"idx": idx_end - 1, "mode": "mid", "expect": p, "carry_step": p,
                            "carry_upto": p, "skipped": p - 1}
                elif pt_prev[0] == p - 1 and p - 1 >= 1:
                    plan = {"idx": idx_end - 1, "mode": "start", "expect": p - 1, "carry_step": p,
                            "carry_upto": p - 1, "skipped": p - 1}
                elif p - 1 == 0:
                    return None          # p = 1 tek substep: basindan = sifirdan
        try:
            self.settings.CurrentRestartPoint = plan["idx"]
        except Exception:
            return None
        return plan

    def _key(self, line):
        m = RestartHelper.RE_RMODIF.match(line)
        if m:
            return (u"rmodif", m.group(1).lower(), m.group(2))
        m = RestartHelper.RE_MP.match(line)
        if m:
            return (u"mp", m.group(1).lower())
        return None

    def carry_lines(self, upto):
        """
        Step <= upto snippet'lerindeki Rmodif / MP,MU satirlarinin son degerleri.
        upto+1'deki snippet'lerde zaten tekrar verilenler cikarilir.
        Donen: (satirlar, uyari_listesi)  - uyari: restart'ta yeniden uygulanmayacak komut iceren snippet adlari
        """
        snips = []
        idx = 0
        for ch in self.analysis.Children:
            idx += 1
            try:
                if "CommandSnippet" not in ch.GetType().ToString():
                    continue
                if ch.Name.startswith(SNIP_PREFIX):
                    continue
                st = rs_snippet_step(ch)
                if st is None:
                    continue
                snips.append((st, idx, ch))
            except Exception:
                pass
        snips.sort(key=lambda t: (t[0], t[1]))
        entries  = {}
        order    = []
        explicit = set()
        warn     = []
        for (st, i, ch) in snips:
            try:
                text = ch.Input
            except Exception:
                continue
            for raw in text.split(u"\n"):
                line = raw.strip()
                if not line:
                    continue
                key = self._key(line)
                if st == upto + 1:
                    if key:
                        explicit.add(key)
                    continue
                if st > upto:
                    continue
                if key:
                    entries[key] = line
                    if key not in order:
                        order.append(key)
                elif not (line.startswith(u"!") or line.upper().startswith(u"/COM")):
                    if ch.Name not in warn:
                        warn.append(ch.Name)
        return [entries[k] for k in order if k not in explicit], warn

    def solve(self, cancelled=None):
        """cancelled verilirse cozum kesilebilir (Stop -> solver abort edilir); info["cancelled"] doner."""
        t0 = time.time()
        err = None
        was_cancelled = False
        try:
            if cancelled is not None:
                err, was_cancelled = solve_interruptible(self.analysis, cancelled, self.workdir)
            else:
                self.analysis.Solve(True)
        except Exception as ex:
            err = unicode(str(ex))
        wall = time.time() - t0
        try:
            p = os.path.join(self.workdir, u"file.abt")
            if os.path.exists(p):
                os.remove(p)
        except Exception:
            pass
        info = rs_parse_solve_out(self.workdir) if self.workdir else {"restart_from": None}
        info["wall"]      = wall
        info["error"]     = err
        info["cancelled"] = was_cancelled
        return info

def probe_value(fr, dir_code):
    """Evaluate ETMEZ; sadece degeri okur."""
    if dir_code == u"Total":
        q = fr.Total
    elif dir_code[0] == u"X":
        q = fr.XAxis
    elif dir_code[0] == u"Y":
        q = fr.YAxis
    else:
        q = fr.ZAxis
    return abs(float(q.Value))

def read_probes_once(solution, its):
    """
    Tum probe'larin DisplayTime'ini kendi step sonuna ayarlar, SONRA TEK BIR
    Solution.EvaluateAllResults() cagirir ve degerleri okur. {item idx: |kuvvet|}
    """
    for it in its:
        try:
            it["probe"].DisplayTime = make_time(it["end_time"])
        except Exception:
            pass
    try:
        solution.EvaluateAllResults()
    except Exception:
        pass
    out = {}
    for it in its:
        try:
            out[it["idx"]] = probe_value(it["probe"], it["dir"])
        except Exception:
            # sadece bu probe okunamadiysa onu ayrica evaluate et
            try:
                it["probe"].EvaluateAllResults()
            except Exception:
                pass
            out[it["idx"]] = probe_value(it["probe"], it["dir"])
    return out

def run_iteration(analysis, items, max_iter, use_cut, final_solve, use_restart, mid_restart, prog, cancelled):
    """Restart kullaniliyorsa ayarlar kaydedilir ve is bitince kullanicinin buldugu gibi geri yuklenir."""
    RS_STATS["restart"] = 0
    RS_STATS["scratch"] = 0
    RS_STATS["skipped"] = 0
    del RS_WARN[:]
    workdir = get_workdir(analysis)
    rh = None
    if use_restart and workdir:
        rh = RestartHelper(analysis, workdir, mid=bool(mid_restart))
        try:
            rh.setup()
        except Exception:
            try:
                rh.restore()
            except Exception:
                pass
            rh = None
    try:
        return _run_iteration_core(analysis, items, max_iter, use_cut, final_solve, rh, bool(mid_restart), prog, cancelled)
    finally:
        try:
            remove_stale_snippets(analysis)
        except Exception:
            pass
        if rh is not None:
            try:
                rh.restore()
            except Exception:
                pass

def _run_iteration_core(analysis, items, max_iter, use_cut, final_solve, rh, mid_restart, prog, cancelled):
    """
    ASAMALI iterasyon: preload'lar step sirasina gore cozulur. Once en kucuk step'teki satirlar
    (kendi icinde birlikte) tutturulur, sonra bir sonraki step'e gecilir:  2 -> 3 -> 4 ...
    Her asamada cozum o step'te durdurulur; restart bir onceki step'in sonundan baslar.
    """
    settings    = analysis.AnalysisSettings
    solution    = analysis.Solution
    total_steps = get_step_count(analysis)
    steps       = sorted(set([it["step"] for it in items]))
    ctrl_max    = steps[-1]
    last_cut    = bool(use_cut and ctrl_max < total_steps)
    err         = None
    workdir     = get_workdir(analysis)

    if use_cut and workdir is None and steps[0] < total_steps:
        return 0, u"Solver working directory could not be determined (analysis.WorkingDir)."

    remove_stale_snippets(analysis)

    prog.update(u"Preparing...", u"")
    end_times  = get_step_end_times(settings, steps)
    used_names = []
    for i, it in enumerate(items):
        it["idx"] = i
        try:
            it["cs_id"] = int(it["cs_obj"].CoordinateSystemID)
        except:
            it["cs_id"] = 0
        it["probe"]     = None
        it["end_time"]  = end_times[it["step"]]
        it["x"]         = it["x0"]
        it["written_x"] = None
        it["solved_x"]  = None
        it["hist"]      = []
        it["converged"] = False
        it["F"]         = None
        it["F_final"]   = None

    titles = []
    for n_i, it in enumerate(items):
        t = str(n_i + 1) + u".  " + it["contact"].Name
        if ckey(it["fcontact"]) != ckey(it["contact"]):
            t += u"   ->   force @ " + it["fcontact"].Name
        t += u"   [step " + str(it["step"]) + u",  target " + fmt_val(it["target"] / 1000.0) + u" kN +/-" + fmt_val(it["tol"]) + u"%]"
        titles.append(t)
    try:
        prog.enable_log(titles)
    except:
        pass

    iters_done = 0
    snips      = []
    carry      = None
    chain_ok   = False
    try:
        for si, s in enumerate(steps):
            if err is not None or cancelled[0]:
                break
            stage_items = [it for it in items if it["step"] == s]
            cut_stage   = bool(use_cut and s < total_steps)

            # bu asamanin probe'lari (onceki asamalarinkiler yerinde kalir)
            for it in stage_items:
                pname = u"Preload_" + it["fcontact"].Name
                if pname in used_names:
                    pname = pname + u" #" + str(it["idx"] + 1)
                used_names.append(pname)
                it["probe"] = create_probe(solution, pname, it["fcontact"], it["cs_obj"])
            if cut_stage:
                snips = add_read_snippets(analysis, stage_items, s)

            for k in range(1, max_iter + 1):
                if cancelled[0]:
                    break
                iters_done += 1
                tag = (u"Step " + str(s) + u"  (" + str(si + 1) + u"/" + str(len(steps)) +
                       u")  -  iteration " + str(k) + u"/" + str(max_iter))

                # 1) bu asamadaki satirlarin komutlarini yaz
                prog.update(tag + u"  -  writing commands", u"")
                for it in stage_items:
                    if it["written_x"] is None or abs(it["written_x"] - it["x"]) > 0.0:
                        set_cmd_value(analysis, u"interference", it["step"], it["cid"], fmt_val(it["x"]))
                        it["written_x"] = it["x"]

                # 2) restart plani: onceki asamalar bitti, hep bir onceki step'in sonundan basla
                expected   = None
                skipped_n  = 0
                check_step = None
                where      = u"From scratch"
                if rh is not None:
                    if (si == 0 and k == 1) or not chain_ok:
                        rh.scratch()
                    else:
                        # step ortasindan restart yalnizca bir onceki step'te preload satiri varsa (sonuc kontrol edilebilir)
                        use_mid = bool(mid_restart and len([it for it in items if it["step"] == s - 1]) > 0)
                        plan = rh.choose(s, use_mid)
                        if plan is not None:
                            expected  = plan["expect"]
                            skipped_n = plan["skipped"]
                            lines, warn = rh.carry_lines(plan["carry_upto"])
                            for w in warn:
                                if w not in RS_WARN:
                                    RS_WARN.append(w)
                            if lines:
                                carry = rs_add_carry(analysis, plan["carry_step"], lines)
                            if plan["mode"] in ("mid", "start"):
                                check_step = s - 1
                            if plan["mode"] == "mid":
                                where = (u"Restart from step " + str(s - 1) + u", one substep before its end (steps 1.." +
                                         str(s - 2) + u" not re-solved)")
                            elif plan["mode"] == "start":
                                where = u"Restart from the start of step " + str(s - 1) + u" (it has a single substep)"
                            else:
                                where = (u"Restart from the end of step " + str(expected) +
                                         u" (steps 1.." + str(expected) + u" not re-solved)")
                        else:
                            rh.scratch()

                # 3) coz
                if cut_stage:
                    prog.update(tag + u"  -  solving",
                                where + u"  |  solver stops after step " + str(s) + u" of " + str(total_steps))
                    clean_iter_files(workdir, items)
                else:
                    prog.update(tag + u"  -  solving", where + u"  |  full analysis")
                solve_err = None
                if rh is not None:
                    info = rh.solve(cancelled)
                    if info.get("error"):
                        solve_err = u"Solve error: " + info["error"]
                    if cancelled[0] or info.get("cancelled"):
                        break
                    rf = info.get("restart_from")
                    if rf is not None and rf != expected:
                        err = (u"Unexpected restart: solver restarted from load step " + unicode(str(rf)) +
                               u", expected " + unicode(str(expected)) + u".")
                        break
                    if rf is not None:
                        RS_STATS["restart"] += 1
                        RS_STATS["skipped"] += skipped_n
                    else:
                        RS_STATS["scratch"] += 1
                else:
                    solve_err, was_canc = solve_interruptible(analysis, cancelled, workdir)
                    if solve_err:
                        solve_err = u"Solve error: " + solve_err
                    if cancelled[0] or was_canc:
                        break

                # 4) oku: TEK evaluate (durdurulan cozumde de tamamlanan step'ler icin)
                try:
                    if not cut_stage:
                        if solve_err is None and not solve_ok(analysis):
                            solve_err = u"Solution did not finish successfully (status: " + unicode(str(analysis.Solution.Status)) + u")"
                        if solve_err is not None:
                            err = solve_err
                            break
                    prog.update(tag + u"  -  evaluating", u"")
                    check_items = []
                    if check_step is not None:
                        check_items = [it for it in items if it["step"] == check_step and len(it["hist"]) > 0]
                    fmap = read_probes_once(solution, stage_items + check_items)
                    for it in check_items:
                        old = it["F"]
                        new = fmap[it["idx"]]
                        if abs(new - old) > 1e-4 * max(abs(old), 1.0):
                            err = (u"Restart consistency check failed: item " + str(it["idx"] + 1) + u" (step " +
                                   str(it["step"]) + u") was " + fmt_val(old / 1000.0) + u" kN before and " +
                                   fmt_val(new / 1000.0) + u" kN after the restart re-solved that step. "
                                   u"The restart did not reproduce the earlier result. Untick 'Restart from the substep "
                                   u"before the end of the previous step' (or 'Use Ansys restart') and run again.")
                            break
                    if err is not None:
                        break
                    for it in stage_items:
                        F = fmap[it["idx"]]
                        it["F"] = F
                        it["solved_x"] = it["x"]
                        it["hist"].append((it["x"], F))
                    chain_ok = True
                except Exception as ex2:
                    err = u"Could not read results: " + unicode(str(ex2))
                    if solve_err is not None:
                        err = err + u"  |  " + solve_err
                    break

                if carry is not None:
                    try:
                        carry.Delete()
                    except Exception:
                        pass
                    carry = None

                # 5) bu asamanin yakinsama kontrolu + yeni deger
                all_ok = True
                for it in stage_items:
                    it["converged"] = within_tol(it["F"], it["target"], it["tol"])
                    if not it["converged"]:
                        all_ok = False
                    prog.log_trial(it["idx"],
                                   u"Trial " + str(len(it["hist"])) + u":   preload " + fmt_val(it["x"]) + u" mm   ->   force " +
                                   fmt_val(it["F"] / 1000.0) + u" kN   [" + (u"OK" if it["converged"] else u"out of tolerance") + u"]",
                                   it["converged"])
                if all_ok or k == max_iter:
                    break
                for it in stage_items:
                    if not it["converged"]:
                        it["x"] = next_value(it["hist"], it["target"])

            # asama tutturulamadiysa DUR (sonraki asamalara gecme)
            if err is None and not cancelled[0]:
                bad = [it for it in stage_items if not it["converged"]]
                if bad:
                    err = (u"Step " + str(s) + u": the preload did not reach the target within " + str(max_iter) +
                           u" iterations. Not converged: " +
                           u", ".join([str(it["idx"] + 1) + u" (" + fmt_val(it["F"] / 1000.0) + u" kN, target " +
                                       fmt_val(it["target"] / 1000.0) + u" kN)" for it in bad]) +
                           u". The remaining steps were not processed; the last tried interference values are kept in the commands.")

            # asama bitti (tutturuldu / max iterasyon / iptal): bu asamanin durdurma snippet'ini kaldir
            delete_snippets(snips)
            snips = []
            clean_iter_files(workdir, items)
    finally:
        delete_snippets(snips)
        clean_iter_files(workdir, items)

    if cancelled[0]:
        return iters_done, None

    # 6) bitis: EN SON cozulen interference degerleri ile devam
    if err is None:
        for it in items:
            if len(it["hist"]) == 0:
                continue
            it["x"] = it["hist"][-1][0]
            it["F"] = it["hist"][-1][1]
            it["converged"] = within_tol(it["F"], it["target"], it["tol"])
            if it["written_x"] is None or abs(it["written_x"] - it["x"]) > 0.0:
                set_cmd_value(analysis, u"interference", it["step"], it["cid"], fmt_val(it["x"]))
                it["written_x"] = it["x"]

        have_hist  = (len([it for it in items if len(it["hist"]) > 0]) == len(items))
        full_valid = False
        if last_cut:
            if final_solve and not cancelled[0] and have_hist:
                prog.update(u"Final solve...", u"From scratch, all " + str(total_steps) + u" steps")
                try:
                    if rh is not None:
                        rh.save_last_only()               # final cozum icin her substep'te restart dosyasi gerekmez
                        rh.scratch()                      # final cozum her zaman bastan
                        info = rh.solve(cancelled)
                        if info.get("error"):
                            raise Exception(info["error"])
                        if cancelled[0] or info.get("cancelled"):
                            return iters_done, None
                        if info.get("restart_from") is not None:
                            raise Exception(u"The final solve restarted from load step " +
                                            unicode(str(info.get("restart_from"))) + u" instead of starting from scratch.")
                        RS_STATS["scratch"] += 1
                    else:
                        serr, canc = solve_interruptible(analysis, cancelled, workdir)
                        if serr:
                            raise Exception(serr)
                        if canc or cancelled[0]:
                            return iters_done, None
                    full_valid = solve_ok(analysis)
                    if not full_valid:
                        err = u"Final solve did not finish successfully (status: " + unicode(str(analysis.Solution.Status)) + u")"
                except Exception as ex4:
                    err = u"Final solve error: " + unicode(str(ex4))
        else:
            full_valid = have_hist

        # Final reaction force'lar: TEK evaluate
        if full_valid:
            prog.update(u"Evaluating final reaction forces...", u"")
            try:
                fmap = read_probes_once(solution, items)
                for it in items:
                    it["F_final"] = fmap.get(it["idx"])
            except Exception:
                pass

        # Probe'lara bilgi amacli comment
        if have_hist:
            for it in items:
                try:
                    add_probe_comment(it, iters_done)
                except:
                    pass

    return iters_done, err

    def _on_fill(self, sender, e):
        if len(self._rows) < 2:
            return
        r0 = self._rows[0]
        for r in self._rows[1:]:
            r[3].SelectedIndex = r0[3].SelectedIndex
            r[4].SelectedIndex = r0[4].SelectedIndex
            if r0[5].SelectedIndex >= 0:
                r[5].SelectedIndex = r0[5].SelectedIndex
            r[6].Text = r0[6].Text
            r[7].Text = r0[7].Text
            r[8].Text = r0[8].Text

    def _on_back(self, sender, e):
        self.DialogResult = DialogResult.Retry
        self.Close()

    def _on_start(self, sender, e):
        out = []
        try:
            for (contact, ex_cid, cb_force, cb_step, cb_dir, cb_cs, t_target, t_tol, t_x0) in self._rows:
                target = float(t_target.Text.strip().replace(u",", u".")) * 1000.0   # ekranda kN, icerde N
                tol    = float(t_tol.Text.strip().replace(u",", u"."))
                x0     = float(t_x0.Text.strip().replace(u",", u"."))
                if target <= 0 or tol <= 0 or x0 <= 0:
                    raise ValueError(contact.Name)
                cs_name = None
                if cb_cs.SelectedIndex >= 0:
                    cs_name = unicode(cb_cs.SelectedItem)
                out.append({"contact": contact, "existing_cid": ex_cid,
                            "fcontact": self._all_contacts[cb_force.SelectedIndex][1],
                            "step": cb_step.SelectedIndex + 1,
                            "dir": IterParamForm.DIRS[cb_dir.SelectedIndex],
                            "cs_name": cs_name,
                            "target": target, "tol": tol, "x0": x0})
            mi = int(self.t_maxit.Text.strip())
            if mi < 1:
                raise ValueError(u"max iter")
        except:
            MessageBox.Show(u"Please enter valid positive numbers for target, tolerance, initial value and max iterations.",
                            u"Warning", MessageBoxButtons.OK, MessageBoxIcon.Warning)
            return
        self.items_out   = out
        self.max_iter    = mi
        self.truncate    = self.chk_trunc.Checked
        self.final_solve = self.chk_final.Checked
        self.use_restart = self.chk_restart.Checked
        self.use_mid     = self.chk_mid.Checked
        self.DialogResult = DialogResult.OK
        self.Close()

def run_manual(static_analysis, select_form, cmd_type, cmd_value, step_num, start_cid):
    cancelled_by_user = [False]
    prog       = ProgressForm()
    prog.show_with(cancelled_by_user)
    start_time = DateTime.Now

    total_sel = len(select_form.selected_contacts)
    cid_list, next_cid, reused, created = assign_cids(
        select_form.selected_contacts, start_cid, prog, cancelled_by_user)

    if cancelled_by_user[0]:
        prog.Close()
        MessageBox.Show(u"Operation cancelled by user.", u"Cancelled",
                        MessageBoxButtons.OK, MessageBoxIcon.Warning)
        return

    prog.update(u"Adding solution command to analysis...",
                u"Step " + str(step_num) + u"  |  " + disp_type(cmd_type))

    sol_cmd, is_new = add_or_update_solution_cmd(
        static_analysis, cmd_type, cmd_value, cid_list, step_num)

    elapsed_sec  = int((DateTime.Now - start_time).TotalSeconds)
    duration_str = format_duration(elapsed_sec)
    prog.Close()

    cmd_status = u"New command created" if is_new else u"Existing command updated"
    MessageBox.Show(
        str(total_sel) + u" contact(s) processed.\n" +
        u"  Reused existing CIDs : " + str(reused) + u"\n" +
        u"  New CIDs assigned    : " + str(created) + u"\n" +
        u"  CID range            : cid_" + str(start_cid + 1) +
            u" ... cid_" + str(next_cid) + u"\n\n" +
        u"Solution command : " + cmd_status + u"\n" +
        u"Analysis         : " + static_analysis.Name + u"\n" +
        u"Step             : " + str(step_num) + u"  (By Number)\n" +
        u"Command name     : " + sol_cmd.Name + u"\n" +
        u"Total lines      : " + str(len(sol_cmd.Input.split(u"\n"))) + u"\n\n" +
        u"Duration: " + duration_str,
        u"Done", MessageBoxButtons.OK, MessageBoxIcon.Information)

def run_iteration_mode(model, static_analysis, select_form, par_form, start_cid):
    cancelled_by_user = [False]
    prog       = ProgressForm()
    prog.show_with(cancelled_by_user)
    start_time = DateTime.Now

    items = par_form.items_out

    # Interference kontaklari + force okunan kontaklar (tekrarsiz) icin CID ata
    uniq = []
    keys = []
    for it in items:
        for c in (it["contact"], it["fcontact"]):
            k = ckey(c)
            if k not in keys:
                keys.append(k)
                uniq.append((c, get_existing_cid(c)))
    cid_list, next_cid, reused, created = assign_cids(uniq, start_cid, prog, cancelled_by_user)

    if cancelled_by_user[0]:
        prog.Close()
        MessageBox.Show(u"Operation cancelled by user.", u"Cancelled",
                        MessageBoxButtons.OK, MessageBoxIcon.Warning)
        return
    cmap = dict(zip(keys, cid_list))

    cs_map = dict(get_cs_list(model))
    for it in items:
        it["cid"]    = cmap[ckey(it["contact"])]
        it["fcid"]   = cmap[ckey(it["fcontact"])]
        it["cs_obj"] = cs_map.get(it["cs_name"])

    try:
        iters_done, err = run_iteration(static_analysis, items,
                                        par_form.max_iter, par_form.truncate,
                                        par_form.final_solve, par_form.use_restart, par_form.use_mid,
                                        prog, cancelled_by_user)
    except Exception as ex:
        iters_done = 0
        err = u"Unexpected error: " + unicode(str(ex))

    elapsed_sec = int((DateTime.Now - start_time).TotalSeconds)
    prog.Close()

    lines = []
    for n_i, it in enumerate(items):
        if not it.get("hist"):
            lines.append(str(n_i + 1) + u".  " + it["contact"].Name + u":  no result")
            continue
        status = u"OK" if it["converged"] else u"NOT within tolerance"
        txt = (str(n_i + 1) + u".  " + it["contact"].Name + u"  (interference, cid_" + str(it["cid"]) + u")\n" +
               u"   force read at: " + it["fcontact"].Name + u"  (step " + str(it["step"]) + u", " + it["dir"] + u")\n" +
               u"   interference " + fmt_val(it["x"]) + u" mm   |   target " + fmt_val(it["target"] / 1000.0) +
               u" kN (+/-" + fmt_val(it["tol"]) + u"%)   |   iteration result " + fmt_val(it["F"] / 1000.0) + u" kN  [" + status + u"]")
        if it.get("F_final") is not None:
            txt += u"\n   final reaction force (full solve): " + fmt_val(it["F_final"] / 1000.0) + u" kN"
        lines.append(txt)

    head = u""
    if err is not None:
        head = u"STOPPED: " + err + u"\n\n"
    elif cancelled_by_user[0]:
        head = u"Cancelled by user.\n\n"

    rs_txt = u""
    if par_form.use_restart and (RS_STATS["restart"] + RS_STATS["scratch"]) > 0:
        rs_txt = (u"Restart        : " + str(RS_STATS["restart"]) + u" solve(s) restarted (" + str(RS_STATS["skipped"]) +
                  u" step-solves skipped), " + str(RS_STATS["scratch"]) + u" solve(s) from scratch\n")
        if RS_WARN:
            rs_txt += (u"WARNING        : commands in snippet(s) " + u", ".join(RS_WARN) +
                       u" (steps before the restart step) are NOT re-applied on restart.\n")

    MessageBox.Show(
        head +
        u"Solves run     : " + str(iters_done) + u"   (max " + str(par_form.max_iter) + u" iterations per step)\n" +
        u"Duration       : " + format_duration(elapsed_sec) + u"\n" + rs_txt + u"\n" +
        u"\n\n".join(lines),
        u"Iteration Done", MessageBoxButtons.OK,
        MessageBoxIcon.Information if err is None else MessageBoxIcon.Warning)

model       = ExtAPI.DataModel.Project.Model
connections = model.Connections

contact_tree = build_contact_tree(connections)
total_count  = sum(len(v) for (_, v) in contact_tree)
all_contacts = [(c.Name, c) for (g_name, c_list) in contact_tree for c in c_list]

if total_count == 0:
    MessageBox.Show(u"No contacts found in the analysis.", u"Info",
                    MessageBoxButtons.OK, MessageBoxIcon.Information)
else:
    start_cid       = find_max_existing_cid(connections)
    static_analysis = find_static_structural(model)

    if static_analysis is None:
        MessageBox.Show(u"No Static Structural analysis found in the model.", u"Error",
                        MessageBoxButtons.OK, MessageBoxIcon.Error)
    else:
        step_count = get_step_count(static_analysis)
        cs_names   = [n for (n, c) in get_cs_list(model)]

        type_form   = TypeSelectForm(start_cid, step_count)
        type_result = type_form.ShowDialog()

        while type_result == DialogResult.OK:
            cmd_type  = type_form.selected_type
            cmd_value = type_form.entered_value
            step_num  = type_form.entered_step
            is_iter   = (cmd_type == u"interference" and type_form.preload_mode == u"iteration")

            title_val   = u"iteration" if is_iter else cmd_value
            select_form = ContactSelectForm(contact_tree, cmd_type, title_val, start_cid)

            back_to_type = False
            while True:
                select_result = select_form.ShowDialog()
                if select_result == DialogResult.Retry:
                    back_to_type = True
                    break
                if select_result != DialogResult.OK:
                    break
                if not is_iter:
                    run_manual(static_analysis, select_form, cmd_type, cmd_value, step_num, start_cid)
                    break

                par_form   = IterParamForm(select_form.selected_contacts, step_count, cs_names, all_contacts)
                par_result = par_form.ShowDialog()
                if par_result == DialogResult.Retry:
                    continue
                if par_result == DialogResult.OK:
                    if LocationDefinitionMethod is None:
                        MessageBox.Show(u"LocationDefinitionMethod enum could not be imported; "
                                        u"force reaction probes cannot be created.", u"Error",
                                        MessageBoxButtons.OK, MessageBoxIcon.Error)
                    else:
                        run_iteration_mode(model, static_analysis, select_form, par_form, start_cid)
                break

            if back_to_type:
                type_form   = TypeSelectForm(start_cid, step_count)
                type_result = type_form.ShowDialog()
                continue
            break
