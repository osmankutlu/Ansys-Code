def _style_button(b, kind):
    b.FlatStyle = FlatStyle.Flat
    if kind == u"primary":
        b.BackColor = THEME_BLUE
        b.ForeColor = Color.White
        b.Font      = Font(b.Font, FontStyle.Bold)
    elif kind == u"danger":
        b.BackColor = THEME_STOP
        b.ForeColor = Color.White
        b.Font      = Font(b.Font, FontStyle.Bold)
    else:
        b.BackColor = THEME_GREY
        b.ForeColor = Color.Black

def apply_theme(root):
    for c in root.Controls:
        try:
            if isinstance(c, Button):
                t = c.Text
                if t == u"Stop":
                    _style_button(c, u"danger")
                elif t in PRIMARY_TEXTS:
                    _style_button(c, u"primary")
                else:
                    _style_button(c, u"secondary")
            elif isinstance(c, GroupBox):
                saved = [(k, k.ForeColor) for k in c.Controls]
                c.ForeColor = THEME_BLUE_DARK
                for (k, fc) in saved:
                    k.ForeColor = fc
                apply_theme(c)
            elif isinstance(c, Panel):
                apply_theme(c)
        except:
            pass

def fit_dropdown(combo, font, max_items=12):
    """Acilir liste makul yukseklikte (en cok max_items satir) ve en uzun metne gore genislikte olsun."""
    try:
        combo.MaxDropDownItems = max_items
        combo.IntegralHeight   = True
        w = combo.Width
        for itm in combo.Items:
            tw = TextRenderer.MeasureText(unicode(itm), font).Width + 30
            if tw > w:
                w = tw
        combo.DropDownWidth = min(w, 700)
    except Exception:
        pass

def make_no_scroll_combo():
    combo = ComboBox()
    def on_mouse_wheel(sender, e):
        try:
            e.Handled = True
        except:
            pass
    combo.MouseWheel += on_mouse_wheel
    return combo

def format_duration(elapsed_sec):
    if elapsed_sec < 60:
        return str(elapsed_sec) + u" sec"
    else:
        mins = elapsed_sec // 60
        secs = elapsed_sec - mins * 60
        return str(mins) + u" min " + str(secs) + u" sec"

def disp_type(cmd_type):
    if cmd_type == u"friction":
        return u"FRICTION"
    return u"PRELOAD"

def fmt_val(x):
    t = (u"%.8f" % x).rstrip(u"0").rstrip(u".")
    if t == u"" or t == u"-":
        t = u"0"
    return t

# Kontak altindaki mevcut cid numarasini bul
# ---------------------------------------------------------------
def get_existing_cid(contact):
    for child in contact.Children:
        if "CommandSnippet" not in child.GetType().ToString():
            continue
        try:
            matches = re.findall(r"cid_(\d+)\s*=\s*cid", child.Input)
            if matches:
                return int(matches[0])
        except:
            pass
    return None

# ---------------------------------------------------------------
# Tum kontaklardaki max cid numarasini bul
# ---------------------------------------------------------------
def find_max_existing_cid(connections):
    max_cid = 0
    for conn_group in connections.Children:
        for contact in conn_group.Children:
            if "ContactRegion" not in contact.GetType().ToString():
                continue
            cid = get_existing_cid(contact)
            if cid is not None and cid > max_cid:
                max_cid = cid
    return max_cid

def build_contact_tree(connections):
    tree = []
    for conn_group in connections.Children:
        group_contacts = []
        for contact in conn_group.Children:
            if "ContactRegion" in contact.GetType().ToString():
                group_contacts.append(contact)
        if group_contacts:
            tree.append((conn_group.Name, group_contacts))
    return tree

def make_cmd_line(cmd_type, cid_n, cmd_value):
    if cmd_type == u"friction":
        return u"MP,MU,cid_" + str(cid_n) + u"," + cmd_value
    else:
        return u"Rmodif,cid_" + str(cid_n) + u",10," + cmd_value

def cmd_prefix(cmd_type):
    if cmd_type == u"friction":
        return u"MP,MU,"
    else:
        return u"Rmodif,"

def cmd_line_key(cmd_type, cid_n):
    # Belirli bir cid satirini tanimak icin (cid_1 ile cid_10 karismasin diye sondaki virgul var)
    if cmd_type == u"friction":
        return (u"MP,MU,cid_" + str(cid_n) + u",").upper()
    else:
        return (u"Rmodif,cid_" + str(cid_n) + u",10,").upper()

# ---------------------------------------------------------------
# Mevcut ayni tip+step command'i bul
# ---------------------------------------------------------------
def find_existing_solution_cmd(analysis, cmd_type, step_num):
    prefix = cmd_prefix(cmd_type)
    for child in analysis.Children:
        if "CommandSnippet" not in child.GetType().ToString():
            continue
        try:
            if child.Name.startswith(u"PRELOAD_ITER_STEP"):
                continue
        except:
            pass
        try:
            snum = None
            try:
                snum = int(child.StepNumber)
            except:
                pass
            try:
                snum = int(child.InternalObject.StepNumber)
            except:
                pass
            if snum != step_num:
                continue
            if prefix.upper() in child.Input.upper():
                return child
        except:
            pass
    return None

# ---------------------------------------------------------------
# Mevcut command'e ekle veya yeni olustur
# ---------------------------------------------------------------
def add_or_update_solution_cmd(analysis, cmd_type, cmd_value, cid_list, step_num):
    existing  = find_existing_solution_cmd(analysis, cmd_type, step_num)
    new_lines = [make_cmd_line(cmd_type, cid_n, cmd_value) for cid_n in cid_list]

    if existing is not None:
        current_lines = [l for l in existing.Input.split(u"\n") if l.strip()]
        for line in new_lines:
            if line not in current_lines:
                current_lines.append(line)
        existing.Input = u"\n".join(current_lines)
        return existing, False
    else:
        cmd       = analysis.AddCommandSnippet()
        cmd.Name  = u"APDL_" + cmd_type.upper() + u"_STEP" + str(step_num)
        cmd.Input = u"\n".join(new_lines)
        cmd.StepSelectionMode = SequenceSelectionType.ByNumber
        cmd.StepNumber        = step_num
        return cmd, True

# ---------------------------------------------------------------
# Iterasyon icin: belirli cid satirinin DEGERINI degistir
# (satir varsa yerinde guncelle, yoksa ekle, command yoksa olustur)
# ---------------------------------------------------------------
def set_cmd_value(analysis, cmd_type, step_num, cid_n, value_str):
    existing = find_existing_solution_cmd(analysis, cmd_type, step_num)
    if existing is None:
        add_or_update_solution_cmd(analysis, cmd_type, value_str, [cid_n], step_num)
        return
    key       = cmd_line_key(cmd_type, cid_n)
    new_line  = make_cmd_line(cmd_type, cid_n, value_str)
    lines     = [l.strip() for l in existing.Input.split(u"\n") if l.strip()]
    found     = False
    for i in range(len(lines)):
        if lines[i].upper().startswith(key):
            lines[i] = new_line
            found = True
    if not found:
        lines.append(new_line)
    existing.Input = u"\n".join(lines)

# ---------------------------------------------------------------
# CID atama (hem manuel hem iterasyon modu kullanir)
# ---------------------------------------------------------------
def assign_cids(selected, start_cid, prog, cancelled):
    next_cid = start_cid
    cid_list = []
    reused   = 0
    created  = 0
    total    = len(selected)
    for idx, (contact, existing_cid) in enumerate(selected):
        if cancelled[0]:
            break
        prog.update(u"Adding CID commands...  (" + str(idx + 1) + u"/" + str(total) + u")",
                    contact.Name)
        if existing_cid is not None:
            cid_list.append(existing_cid)
            reused += 1
        else:
            next_cid += 1
            cid_list.append(next_cid)
            cmd_c       = contact.AddCommandSnippet()
            cmd_c.Name  = u"Contact_Command_" + str(next_cid)
            cmd_c.Input = u"cid_" + str(next_cid) + u" = cid"
            created += 1
    return cid_list, next_cid, reused, created

SNIP_PREFIX = u"PRELOAD_ITER_STEP"

def build_abort_block():
    # /EOF: giris dosyasini burada bitirir; kontrol step'inden sonrasi cozulmez (restart-safe)
    return [u"! --- preload iteration: stop the solver here", u"/EOF"]

def remove_stale_snippets(analysis):
    for ch in list(analysis.Children):
        try:
            if ch.Name.startswith(SNIP_PREFIX):
                ch.Delete()
        except:
            pass

def add_read_snippets(analysis, items, ctrl_max):
    # Kontrol step'inden sonraki step'in basinda cozucuyu durdur (abort dosyasi)
    cmd       = analysis.AddCommandSnippet()

    cmd.Input = u"\n".join(build_abort_block())
    cmd.StepSelectionMode = SequenceSelectionType.ByNumber
    cmd.StepNumber        = ctrl_max + 1
    return [cmd]

def trial_rows(it):
    """Excel tablosu satirlari: baslik + her deneme icin bir satir. (satirlar, ok listesi)"""
    rows = [[u"SUMMARY", u"Interference [mm]", u"Step", u"Applied Contact", u"Read Contact",
             u"Value [kN]", u"Tolerance", u"Situation"]]
    oks = []
    for n, (x, F) in enumerate(it["hist"]):
        ok = tol_ok(it, F)
        oks.append(ok)
        rows.append([u"Trial " + str(n + 1), round(x, 6), it["step"], it["contact"].Name, it["fcontact"].Name,
                     round(F / 1000.0, 3), tol_cell(it), u"OK" if ok else u"Not OK"])
    return rows, oks

def html_esc(t):
    return unicode(t).replace(u"&", u"&amp;").replace(u"<", u"&lt;").replace(u">", u"&gt;")

def rows_to_html(rows, oks):
    """Excel kullanilamazsa ayni gorunumde basit HTML tablo."""
    h = [u"<table style='border-collapse:collapse;font-family:Calibri,Arial;font-size:11pt'>"]
    for r, row in enumerate(rows):
        if r == 0:
            bg, wt = u"#D9D9D9", u"bold"
        elif oks[r - 1]:
            bg, wt = u"#A9D08E", u"normal"
        else:
            bg, wt = u"#FFFFFF", u"normal"
        h.append(u"<tr>")
        for v in row:
            h.append(u"<td style='border:1px solid #808080;padding:2px 8px;background-color:" + bg +
                     u";font-weight:" + wt + u"'>" + html_esc(v) + u"</td>")
        h.append(u"</tr>")
    h.append(u"</table>")
    return u"".join(h)

def read_clipboard_html():
    for i in range(15):
        try:
            if Clipboard.ContainsText(TextDataFormat.Html):
                return Clipboard.GetText(TextDataFormat.Html)
        except Exception:
            pass
        Application.DoEvents()

    return None

def inline_excel_html(html):
    """Excel'in panodaki HTML'inden tabloyu alir; <style> sinif tanimlarini satir ici stile cevirir."""
    m = re.search(r"<!--StartFragment-->(.*?)<!--EndFragment-->", html, re.S)
    if m:
        frag = m.group(1)
    else:
        i = html.lower().index("<table")
        frag = html[i:]
    rules = {}
    for sm in re.finditer(r"<style[^>]*>(.*?)</style>", html, re.S | re.I):
        for mm in re.finditer(r"\.([A-Za-z0-9_]+)\s*\{([^}]*)\}", sm.group(1)):
            decls = [d.strip() for d in mm.group(2).split(u";") if d.strip() and not d.strip().lower().startswith(u"mso-")]
            rules[mm.group(1)] = u";".join(decls)
    def fix_tag(mt):
        tag = mt.group(0)
        cm = re.search(r"class=['\"]?([A-Za-z0-9_]+)['\"]?", tag)
        if not cm:
            return tag
        decl = rules.get(cm.group(1), u"")
        tag = tag.replace(cm.group(0), u"", 1)
        sm2 = re.search(r"style=(['\"])(.*?)\1", tag, re.S)
        if sm2:
            q = sm2.group(1)
            tag = tag.replace(sm2.group(0), u"style=" + q + decl + u";" + sm2.group(2) + q, 1)
        else:
            tag = tag[:-1] + u" style='" + decl + u"'>"
        return tag
    frag = re.sub(r"<(?:td|th|tr|table)\b[^>]*>", fix_tag, frag, flags=re.I)
    frag = re.sub(r"<!\[[^>]*\]>", u"", frag)
    if u"<table" not in frag.lower():
        raise Exception(u"no table in clipboard html")
    return frag

def excel_tables(items):
    """
    Her kontak icin Excel'de tablo olusturur, kopyalar ve panodaki HTML'i alir.
    Excel dosyasi hic kaydedilmez; is bitince kapatilir (silinecek dosya kalmaz). {item idx: html}
    """
    t = System.Type.GetTypeFromProgID("Excel.Application")
    if t is None:
        raise Exception(u"Excel is not installed")
    xl = System.Activator.CreateInstance(t)
    wb = None
    out = {}
    try:
        xl.Visible = False
        xl.DisplayAlerts = False
        wb = xl.Workbooks.Add()
        ws = wb.Worksheets[1]
        for it in items:
            if not it["hist"]:
                continue
            rows, oks = trial_rows(it)
            ws.Cells.Clear()
            for r, row in enumerate(rows):
                for c, v in enumerate(row):
                    ws.Cells[r + 1, c + 1].Value2 = v
            n = len(rows)
            ws.Range("A1:H" + str(n)).Borders.LineStyle = 1
            ws.Range("A1:H1").Font.Bold = True
            ws.Range("A1:H1").Interior.Color = 14277081
            for r, ok in enumerate(oks):
                if ok:
                    ws.Range("A" + str(r + 2) + ":H" + str(r + 2)).Interior.Color = 9359529
            ws.Columns("A:H").AutoFit()
            ws.Range("A1:H" + str(n)).Copy()
            html = read_clipboard_html()
            if html:
                try:
                    out[it["idx"]] = inline_excel_html(html)
                except Exception:
                    pass
    finally:
        try:
            xl.CutCopyMode = False
        except Exception:
            pass
        try:
            if wb is not None:
                wb.Close(False)
        except Exception:
            pass
        try:
            xl.Quit()
        except Exception:
            pass
        for o in (wb, xl):
            try:
                if o is not None:
                    Marshal.ReleaseComObject(o)
            except Exception:
                pass
        try:
            Clipboard.Clear()
        except Exception:
            pass
    return out

def build_comment_tables(items):
    res = {}
    try:
        res = excel_tables(items)
    except Exception:
        res = {}
    for it in items:
        if it["hist"] and it["idx"] not in res:
            rows, oks = trial_rows(it)
            res[it["idx"]] = rows_to_html(rows, oks)
    return res

    if cs_obj is not None:
        try:
            fr.Orientation = cs_obj

    return fr

def within_tol(F, target, tol_pct, mode=u"both"):
    """mode: both = +/-tol ; lower = [hedef*(1-tol), hedef] ; upper = [hedef, hedef*(1+tol)]"""
    d    = F - target
    band = abs(target) * tol_pct / 100.0
    eps  = 1e-9 * max(abs(target), 1.0)
    if mode == u"lower":
        return (-band - eps) <= d <= eps
    if mode == u"upper":
        return -eps <= d <= (band + eps)
    return abs(d) <= band + eps

def tol_ok(it, F):
    return within_tol(F, it["target"], it["tol"], it.get("tol_mode", u"both"))

def tol_text(it):
    m = it.get("tol_mode", u"both")
    if m == u"lower":
        return u"-" + fmt_val(it["tol"]) + u"%"
    if m == u"upper":
        return u"+" + fmt_val(it["tol"]) + u"%"
    return u"+/-" + fmt_val(it["tol"]) + u"%"

def tol_cell(it):
    m = it.get("tol_mode", u"both")
    if m == u"lower":
        return u"-" + fmt_val(it["tol"]) + u"%"
    if m == u"upper":
        return u"+" + fmt_val(it["tol"]) + u"%"
    return fmt_val(it["tol"]) + u"%"

def aim_force(it):
    """Iterasyonun hedefledigi kuvvet: tek tarafli toleransta kabul bandinin ortasi."""
    m = it.get("tol_mode", u"both")
    if m == u"lower":
        return it["target"] * (1.0 - it["tol"] / 200.0)
    if m == u"upper":
        return it["target"] * (1.0 + it["tol"] / 200.0)
    return it["target"]

def next_value(hist, target):
    # hist: [(x, F), ...]   son nokta = hist[-1]
    x1, f1 = hist[-1]
    prev = None
    for k in range(len(hist) - 2, -1, -1):
        if abs(hist[k][0] - x1) > 1e-12 * max(1.0, abs(x1)):
            prev = hist[k]
            break
    xn = None
    if prev is not None and abs(f1 - prev[1]) > 1e-12:
        # sekant: interpolasyon / ekstrapolasyon
        xn = x1 + (target - f1) * (x1 - prev[0]) / (f1 - prev[1])
    else:
        # oran-orantı
        if abs(f1) < 1e-9:
            xn = x1 * 2.0
        else:
            xn = x1 * target / f1
    if xn <= 0:
        xn = x1 * 0.5
    if xn > 5.0 * x1:
        xn = 5.0 * x1
    if xn < 0.1 * x1:
        xn = 0.1 * x1
    return xn

def set_enum_name(obj, prop, name):
    """Enum ozelligini isimle atar; zaten ayni ise dokunmaz."""
    try:
        cur = getattr(obj, prop)
        if unicode(str(cur)).replace(" ", "").lower() == name.lower():
            return True
        setattr(obj, prop, System.Enum.Parse(cur.GetType(), name))
        return True
    except Exception:
        return False

def rs_is_solving(analysis):
    for obj in (analysis.Solution, analysis):
        try:
            if "solving" in str(obj.ObjectState).lower():
                return True
        except Exception:
            pass
    try:
        if "solving" in str(analysis.Solution.Status).lower():
            return True
    except Exception:
        pass
    return False

def rs_abort_dirs(workdir):
    dirs = []
    if workdir:
        dirs.append(workdir)
        # solve.out basligindaki scratch klasoru (varsa) - dosyalar orada okunur
        try:
            f = open(os.path.join(workdir, u"solve.out"), "r")
            try:
                n = 0
                for line in f:
                    n += 1
                    if n > 600:
                        break
                    if "_WB_PROJECTSCRATCH_DIR" in line:
                        m = re.search(r"=\s*['\"]?([^'\"\r\n]+)", line)
                        if m:
                            d = m.group(1).strip()
                            if os.path.isdir(d) and d not in dirs:
                                dirs.append(d)
                        break
            finally:
                f.close()
        except Exception:
            pass
    return dirs

def rs_send_abort(workdir):
    for d in rs_abort_dirs(workdir):
        try:
            f = open(os.path.join(d, u"file.abt"), "w")
            f.write("nonlinear\n")
            f.close()
        except Exception:
            pass

def rs_clear_abort(workdir):
    for d in rs_abort_dirs(workdir):
        try:
            pth = os.path.join(d, u"file.abt")
            if os.path.exists(pth):
                os.remove(pth)
        except Exception:
            pass

ASYNC_SOLVE = [True]      # asenkron cozum sorun cikarirsa otomatik False olur (bekleyen Solve(True)'ya doner)

def rs_out_path(workdir):
    return os.path.join(workdir, u"solve.out") if workdir else None

def rs_mtime(pth):
    try:
        if pth and os.path.exists(pth):
            return os.path.getmtime(pth)
    except Exception:
        pass
    return 0.0

def rs_out_finished(pth, m0):
    """solve.out m0'dan sonra guncellenmis ve bitis ozeti ('Elapsed Time (sec)') iceriyor mu?"""
    try:
        if rs_mtime(pth) <= m0:
            return False
        f = open(pth, "rb")
        try:
            f.seek(0, 2)
            size = f.tell()
            f.seek(max(0, size - 8000))
            tail = f.read()
        finally:
            f.close()
        return "Elapsed Time (sec)" in tail
    except Exception:
        return False

def rs_solver_errors(workdir):
    """solve.out'taki ilk hata satirlari (varsa) - hata mesajina eklenir."""
    pth = rs_out_path(workdir)
    if not pth:
        return u""
    try:
        f = open(pth, "r")
        try:
            lines = f.read().splitlines()
        finally:
            f.close()
    except Exception:
        return u""
    hits = [l.strip() for l in lines if ("*** ERROR ***" in l or "FATAL" in l.upper())]
    if not hits:
        return u""
    return u"  |  solver output: " + u" / ".join([unicode(h, "latin-1") for h in hits[:3]])

def solve_blocking(analysis):
    try:
        analysis.Solve(True)
        return None
    except Exception as ex:
        return unicode(str(ex))

def solve_interruptible(analysis, cancelled, workdir):
    """
    Cozumu ASENKRON baslatir (Solve(False)) ve beklerken arayuzu isler; boylece Stop
    cozum sirasinda da calisir: Stop'a basilirsa solver'a abort dosyasi (file.abt) yazilir.
    Bitis, SOLVE.OUT'un guncellenip bitis ozetini icermesi ile dogrulanir (durum bilgisi tek basina yetmez).
    Asenkron cozum calismazsa otomatik olarak bekleyen Solve(True)'ya doner.
    Cozucu hic calismadiysa HATA doner (sessizce 0 kuvvet okunmaz).
    Donen: (hata metni veya None, iptal edildi mi)
    """
    pth = rs_out_path(workdir)
    m0  = rs_mtime(pth)
    if (not ASYNC_SOLVE[0]) or (not rs_state_readable(analysis)) or pth is None:
        err = solve_blocking(analysis)
        if err is None and pth is not None and rs_mtime(pth) <= m0:
            err = u"The solver did not run (solve.out was not updated)."
        return err, False
    t0 = time.time()
    try:
        analysis.Solve(False)
    except Exception as ex:
        ASYNC_SOLVE[0] = False
        if u"async" not in RS_WARN:
            RS_WARN.append(u"async solve unavailable (" + unicode(str(ex))[:80] + u"): Stop works only between solves")
        err = solve_blocking(analysis)
        if err is None and rs_mtime(pth) <= m0:
            err = u"The solver did not run (solve.out was not updated)."
        return err, False
    seen_busy  = False
    abort_sent = False
    abort_t    = 0.0
    idle_since = None
    while True:
        Application.DoEvents()
        Thread.Sleep(200)
        busy = rs_is_solving(analysis)
        now  = time.time()
        if busy:
            seen_busy  = True
            idle_since = None
        elif idle_since is None:
            idle_since = now
        if cancelled[0] and not abort_sent:
            rs_send_abort(workdir)
            abort_sent = True
            abort_t = now
        fin = rs_out_finished(pth, m0)
        if (not busy) and fin:
            break
        # solve.out guncellenmiyorsa (cozucu baska klasorde calisiyor olabilir) durum bilgisine guven
        if (not busy) and seen_busy and idle_since is not None and (now - idle_since > 30) and rs_mtime(pth) <= m0:
            break
        if (not busy) and seen_busy and idle_since is not None and (now - idle_since > 30) and (now - rs_mtime(pth) > 30):
            break
        if (not seen_busy) and (not fin) and (now - t0 > 120):
            if abort_sent:
                break
            return u"Could not detect that the solver started (the solution never reported 'Solving').", False
        if abort_sent and (now - abort_t > 120):
            break
    if abort_sent:
        rs_clear_abort(workdir)
    if cancelled[0]:
        return None, True
    if rs_mtime(pth) <= m0:
        return u"The solver did not run (solve.out was not updated).", False
    return None, False

def probe_value(fr, dir_code):
    """Evaluate ETMEZ; sadece degeri okur."""
    if dir_code == u"Total":
        q = fr.Total
    elif dir_code[0] == u"X":
        q = fr.XAxis
    elif dir_code[0] == u"Y":
        q = fr.YAxis
    else:
        q = fr.ZAxis
    return abs(float(q.Value))

def read_probes_once(solution, its):
    """
    Tum probe'larin DisplayTime'ini kendi step sonuna ayarlar, SONRA TEK BIR
    Solution.EvaluateAllResults() cagirir ve degerleri okur. {item idx: |kuvvet|}
    """
    for it in its:
        try:
            it["probe"].DisplayTime = make_time(it["end_time"])
        except Exception:
            pass
    try:
        solution.EvaluateAllResults()
    except Exception:
        pass
    out = {}
    for it in its:
        try:
            out[it["idx"]] = probe_value(it["probe"], it["dir"])
        except Exception:
            # sadece bu probe okunamadiysa onu ayrica evaluate et
            try:
                it["probe"].EvaluateAllResults()
            except Exception:
                pass
            out[it["idx"]] = probe_value(it["probe"], it["dir"])
    return out

def run_iteration(analysis, items, max_iter, use_cut, final_solve, prog, cancelled):
    """
    Her iterasyon analizi BASTAN cozer. Mechanical'in kendiliginden restart etmemesi icin
    RestartType gecici olarak 'Manual / Initial' yapilir ve is bitince eski haline getirilir.
    """
    del RS_WARN[:]
    FINAL_STATE["valid"] = False
    FINAL_STATE["state"] = u""
    st = analysis.AnalysisSettings
    saved = {}
    for pn in ("RestartType", "CurrentRestartPoint"):
        try:
            saved[pn] = getattr(st, pn)
        except Exception:
            pass
    set_enum_name(st, "RestartType", "Manual")
    try:
        st.CurrentRestartPoint = 0
    except Exception:
        pass
    try:
        return _run_iteration_core(analysis, items, max_iter, use_cut, final_solve, prog, cancelled)
    finally:
        try:
            remove_stale_snippets(analysis)
        except Exception:
            pass
        for pn in ("RestartType", "CurrentRestartPoint"):
            if pn in saved:
                try:
                    setattr(st, pn, saved[pn])
                except Exception:
                    pass
        # Tum degisiklikler geri alindiktan sonra, tam cozum gecerliyse TUM sonuclari evaluate et
        if FINAL_STATE["valid"]:
            try:
                analysis.Solution.EvaluateAllResults()
            except Exception:
                pass
            try:
                FINAL_STATE["state"] = (unicode(str(analysis.Solution.ObjectState)) + u" / " +
                                        unicode(str(analysis.Solution.Status)))
            except Exception:
                pass

def _run_iteration_core(analysis, items, max_iter, use_cut, final_solve, prog, cancelled):
    """
    ASAMALI iterasyon: preload'lar step sirasina gore cozulur. Once en kucuk step'teki satirlar
    (kendi icinde birlikte) tutturulur, sonra bir sonraki step'e gecilir:  2 -> 3 -> 4 ...
    Her asamada cozum o step'te durdurulur (/EOF) ve her iterasyon analizi bastan cozer.
    """
    settings    = analysis.AnalysisSettings
    solution    = analysis.Solution
    total_steps = get_step_count(analysis)
    steps       = sorted(set([it["step"] for it in items]))
    ctrl_max    = steps[-1]
    last_cut    = bool(use_cut and ctrl_max < total_steps)
    err         = None
    workdir     = get_workdir(analysis)

    if use_cut and workdir is None and steps[0] < total_steps:
        return 0, u"Solver working directory could not be determined (analysis.WorkingDir)."

    remove_stale_snippets(analysis)

    prog.update(u"Preparing...", u"")
    end_times  = get_step_end_times(settings, steps)
    used_names = []
    for i, it in enumerate(items):
        it["idx"] = i
        try:
            it["cs_id"] = int(it["cs_obj"].CoordinateSystemID)
        except:
            it["cs_id"] = 0
        it["probe"]     = None
        it["end_time"]  = end_times[it["step"]]
        it["x"]         = it["x0"]
        it["written_x"] = None
        it["solved_x"]  = None
        it["hist"]      = []
        it["converged"] = False
        it["F"]         = None
        it["F_final"]   = None

    titles = []
    for n_i, it in enumerate(items):
        t = str(n_i + 1) + u".  " + it["contact"].Name
        if ckey(it["fcontact"]) != ckey(it["contact"]):
            t += u"   ->   force @ " + it["fcontact"].Name
        t += u"   [step " + str(it["step"]) + u",  target " + fmt_val(it["target"] / 1000.0) + u" kN " + tol_text(it) + u"]"
        titles.append(t)
    try:
        prog.enable_log(titles)
    except:
        pass

    iters_done = 0
    snips      = []
    try:
        for si, s in enumerate(steps):
            if err is not None or cancelled[0]:
                break
            stage_items = [it for it in items if it["step"] == s]
            cut_stage   = bool(use_cut and s < total_steps)

            # bu asamanin probe'lari (onceki asamalarinkiler yerinde kalir)
            for it in stage_items:
                pname = u"Preload_" + it["fcontact"].Name
                if pname in used_names:
                    pname = pname + u" #" + str(it["idx"] + 1)
                used_names.append(pname)
                it["probe"] = create_probe(solution, pname, it["fcontact"], it["cs_obj"])
            if cut_stage:
                snips = add_read_snippets(analysis, stage_items, s)

            for k in range(1, max_iter + 1):
                if cancelled[0]:
                    break
                iters_done += 1
                tag = (u"Step " + str(s) + u"  (" + str(si + 1) + u"/" + str(len(steps)) +
                       u")  -  iteration " + str(k) + u"/" + str(max_iter))

                # 1) bu asamadaki satirlarin komutlarini yaz
                prog.update(tag + u"  -  writing commands", u"")
                for it in stage_items:
                    if it["written_x"] is None or abs(it["written_x"] - it["x"]) > 0.0:
                        set_cmd_value(analysis, u"interference", it["step"], it["cid"], fmt_val(it["x"]))
                        it["written_x"] = it["x"]

                # 2) coz (her zaman bastan)
                if cut_stage:
                    prog.update(tag + u"  -  solving",
                                u"From the beginning  |  solver stops after step " + str(s) + u" of " + str(total_steps))
                    clean_iter_files(workdir, items)
                else:
                    prog.update(tag + u"  -  solving", u"From the beginning  |  full analysis")
                solve_err, was_canc = solve_interruptible(analysis, cancelled, workdir)
                if solve_err:
                    solve_err = u"Solve error: " + solve_err
                if cancelled[0] or was_canc:
                    break

                # 3) oku: TEK evaluate (durdurulan cozumde de tamamlanan step'ler icin)
                try:
                    if solve_err is None:
                        try:
                            stx = unicode(str(analysis.Solution.Status)).lower()
                        except Exception:
                            stx = u""
                        if cut_stage:
                            if u"fail" in stx:
                                solve_err = u"Solve failed (status: " + stx + u")"
                        elif not solve_ok(analysis):
                            solve_err = u"Solution did not finish successfully (status: " + stx + u")"
                    if solve_err is not None:
                        err = solve_err + rs_solver_errors(workdir)
                        break
                    prog.update(tag + u"  -  evaluating", u"")
                    fmap = read_probes_once(solution, stage_items)
                    for it in stage_items:
                        F = fmap[it["idx"]]
                        it["F"] = F
                        it["solved_x"] = it["x"]
                        it["hist"].append((it["x"], F))
                except Exception as ex2:
                    err = u"Could not read results: " + unicode(str(ex2))
                    break

                # 4) bu asamanin yakinsama kontrolu + yeni deger
                all_ok = True
                for it in stage_items:
                    it["converged"] = tol_ok(it, it["F"])
                    if not it["converged"]:
                        all_ok = False
                    prog.log_trial(it["idx"],
                                   u"Trial " + str(len(it["hist"])) + u":   preload " + fmt_val(it["x"]) + u" mm   ->   force " +
                                   fmt_val(it["F"] / 1000.0) + u" kN   [" + (u"OK" if it["converged"] else u"out of tolerance") + u"]",
                                   it["converged"])
                if all_ok or k == max_iter:
                    break
                for it in stage_items:
                    if not it["converged"]:
                        it["x"] = next_value(it["hist"], aim_force(it))

            # asama tutturulamadiysa DUR (sonraki asamalara gecme)
            if err is None and not cancelled[0]:
                bad = [it for it in stage_items if not it["converged"]]
                if bad:
                    err = (u"Step " + str(s) + u": the preload did not reach the target within " + str(max_iter) +
                           u" iterations. Not converged: " +
                           u", ".join([str(it["idx"] + 1) + u" (" + fmt_val(it["F"] / 1000.0) + u" kN, target " +
                                       fmt_val(it["target"] / 1000.0) + u" kN)" for it in bad]) +
                           u". The remaining steps were not processed; the last tried interference values are kept in the commands.")

            # asama bitti: bu asamanin durdurma snippet'ini kaldir
            delete_snippets(snips)
            snips = []
            clean_iter_files(workdir, items)
    finally:
        delete_snippets(snips)
        clean_iter_files(workdir, items)

    if cancelled[0]:
        return iters_done, None

    # 5) bitis: EN SON cozulen interference degerleri ile devam
    if err is None:
        for it in items:
            if len(it["hist"]) == 0:
                continue
            it["x"] = it["hist"][-1][0]
            it["F"] = it["hist"][-1][1]
            it["converged"] = tol_ok(it, it["F"])
            if it["written_x"] is None or abs(it["written_x"] - it["x"]) > 0.0:
                set_cmd_value(analysis, u"interference", it["step"], it["cid"], fmt_val(it["x"]))
                it["written_x"] = it["x"]

        have_hist  = (len([it for it in items if len(it["hist"]) > 0]) == len(items))
        full_valid = False
        if last_cut:
            if final_solve and not cancelled[0] and have_hist:
                prog.update(u"Final solve...", u"From the beginning, all " + str(total_steps) + u" steps")
                try:
                    prog.update(u"Final solve...", u"Clearing the generated data, then solving from the beginning")
                    cleared = False
                    for obj in (analysis.Solution, analysis):
                        try:
                            obj.ClearGeneratedData()
                            cleared = True
                            break
                        except Exception:
                            pass
                    if not cleared:
                        RS_WARN.append(u"ClearGeneratedData was not available")
                    serr, canc = solve_interruptible(analysis, cancelled, workdir)
                    if serr:
                        raise Exception(serr)
                    if canc or cancelled[0]:
                        return iters_done, None
                    full_valid = solve_ok(analysis)
                    if not full_valid:
                        err = u"Final solve did not finish successfully (status: " + unicode(str(analysis.Solution.Status)) + u")"
                except Exception as ex4:
                    err = u"Final solve error: " + unicode(str(ex4))
        else:
            full_valid = have_hist

        FINAL_STATE["valid"] = bool(full_valid)

        # Final reaction force'lar: TEK evaluate
        if full_valid:
            prog.update(u"Evaluating final reaction forces...", u"")
            try:
                fmap = read_probes_once(solution, items)
                for it in items:
                    it["F_final"] = fmap.get(it["idx"])
            except Exception:
                pass

        # Probe'lara bilgi amacli comment: Excel tablosu (kopyala -> comment'e yapistir), sonra Excel kapatilir
        if have_hist:
            prog.update(u"Creating the summary tables...", u"")
            try:
                htmls = build_comment_tables(items)
            except Exception:
                htmls = {}
            for it in items:
                try:
                    add_probe_comment(it, iters_done, htmls.get(it["idx"]))
                except:
                    pass

    return iters_done, err

    def _on_node_check(self, sender, e):
        if self._updating:
            return
        self._updating = True
        node = e.Node
        if str(node.Tag) == u"GROUP":
            for child in node.Nodes:
                child.Checked = node.Checked
        elif str(node.Tag) == u"CONTACT":
            parent = node.Parent
            if parent is not None:
                all_checked = all(child.Checked for child in parent.Nodes)
                parent.Checked = all_checked
        self._updating = False
        Application.DoEvents()

    def _on_visible(self, sender, e):
        """Form her gosterildiginde (Back ile donuste de) tum gruplar acik olsun."""
        if self.Visible:
            try:
                self.tv.ExpandAll()
                for (node, c, ex) in self._ordered_nodes:
                    if node.Checked:
                        node.EnsureVisible()
                        break
            except Exception:
                pass

    def _on_select_all(self, sender, e):
        self._updating = True
        for grp_node in self._group_nodes:
            grp_node.Checked = True
            for child in grp_node.Nodes:
                child.Checked = True
        self._updating = False

    def _on_clear(self, sender, e):
        self._updating = True
        for grp_node in self._group_nodes:
            grp_node.Checked = False
            for child in grp_node.Nodes:
                child.Checked = False
        self._updating = False

    def _on_back(self, sender, e):
        self.DialogResult = DialogResult.Retry
        self.Close()

    def _on_apply(self, sender, e):
        self.selected_contacts = []
        for (node, contact_obj, existing_cid) in self._ordered_nodes:
            if node.Checked:
                self.selected_contacts.append((contact_obj, existing_cid))
        if len(self.selected_contacts) == 0:
            MessageBox.Show(u"Please select at least one contact.", u"Warning",
                            MessageBoxButtons.OK, MessageBoxIcon.Warning)
            return
        self.DialogResult = DialogResult.OK
        self.Close()

    def _on_start(self, sender, e):
        def num(tb):
            try:
                return float(tb.Text.strip().replace(u",", u"."))
            except Exception:
                return None
        out = []
        problems = []
        for i, (contact, ex_cid, cb_force, cb_step, cb_dir, cb_cs, t_target, t_tol, t_x0) in enumerate(self._rows):
            miss = []
            if cb_force.SelectedIndex < 0:
                miss.append(u"Read force at")
            if cb_step.SelectedIndex < 0:
                miss.append(u"Step")
            if cb_dir.SelectedIndex < 0:
                miss.append(u"Direction")
            if cb_cs.Items.Count > 0 and cb_cs.SelectedIndex < 0:
                miss.append(u"Coordinate system")
            target = num(t_target)
            tol_txt  = t_tol.Text.strip().replace(u",", u".")
            tol_mode = u"both"
            if tol_txt.startswith(u"-"):
                tol_mode = u"lower"
                tol_txt  = tol_txt[1:]
            elif tol_txt.startswith(u"+"):
                tol_mode = u"upper"
                tol_txt  = tol_txt[1:]
            try:
                tol = float(tol_txt)
            except Exception:
                tol = None
            x0     = num(t_x0)
            if target is None or target <= 0:
                miss.append(u"Target [kN]")
            if tol is None or tol <= 0:
                miss.append(u"Tolerance [%]")
            if x0 is None or x0 <= 0:
                miss.append(u"Initial [mm]")
            if miss:
                problems.append(str(i + 1) + u". " + contact.Name + u":  " + u", ".join(miss))
                continue
            cs_name = None
            if cb_cs.SelectedIndex >= 0:
                cs_name = unicode(cb_cs.SelectedItem)
            out.append({"contact": contact, "existing_cid": ex_cid,
                        "fcontact": self._all_contacts[cb_force.SelectedIndex][1],
                        "step": cb_step.SelectedIndex + 1,
                        "dir": IterParamForm.DIRS[cb_dir.SelectedIndex],
                        "cs_name": cs_name,
                        "target": target * 1000.0, "tol": tol, "tol_mode": tol_mode, "x0": x0})   # ekranda kN, icerde N
        mi = None
        try:
            mi = int(self.t_maxit.Text.strip())
        except Exception:
            mi = None
        if mi is None or mi < 1:
            problems.append(u"Max iterations (per step): enter a whole number >= 1")
        if problems:
            MessageBox.Show(u"Please fill in / correct the following:\n\n" + u"\n".join(problems),
                            u"Warning", MessageBoxButtons.OK, MessageBoxIcon.Warning)
            return
        self.items_out   = out
        self.max_iter    = mi
        self.truncate    = self.chk_trunc.Checked
        self.final_solve = self.chk_final.Checked
        self.DialogResult = DialogResult.OK
        self.Close()

def run_manual(static_analysis, select_form, cmd_type, cmd_value, step_num, start_cid):
    cancelled_by_user = [False]
    prog       = ProgressForm()
    prog.show_with(cancelled_by_user)
    start_time = DateTime.Now

    total_sel = len(select_form.selected_contacts)
    cid_list, next_cid, reused, created = assign_cids(
        select_form.selected_contacts, start_cid, prog, cancelled_by_user)

    if cancelled_by_user[0]:
        prog.Close()
        MessageBox.Show(u"Operation cancelled by user.", u"Cancelled",
                        MessageBoxButtons.OK, MessageBoxIcon.Warning)
        return

    prog.update(u"Adding solution command to analysis...",
                u"Step " + str(step_num) + u"  |  " + disp_type(cmd_type))

    sol_cmd, is_new = add_or_update_solution_cmd(
        static_analysis, cmd_type, cmd_value, cid_list, step_num)

    elapsed_sec  = int((DateTime.Now - start_time).TotalSeconds)
    duration_str = format_duration(elapsed_sec)
    prog.Close()

    cmd_status = u"New command created" if is_new else u"Existing command updated"
    MessageBox.Show(
        str(total_sel) + u" contact(s) processed.\n" +
        u"  Reused existing CIDs : " + str(reused) + u"\n" +
        u"  New CIDs assigned    : " + str(created) + u"\n" +
        u"  CID range            : cid_" + str(start_cid + 1) +
            u" ... cid_" + str(next_cid) + u"\n\n" +
        u"Solution command : " + cmd_status + u"\n" +
        u"Analysis         : " + static_analysis.Name + u"\n" +
        u"Step             : " + str(step_num) + u"  (By Number)\n" +
        u"Command name     : " + sol_cmd.Name + u"\n" +
        u"Total lines      : " + str(len(sol_cmd.Input.split(u"\n"))) + u"\n\n" +
        u"Duration: " + duration_str,
        u"Done", MessageBoxButtons.OK, MessageBoxIcon.Information)

def run_iteration_mode(model, static_analysis, select_form, par_form, start_cid):
    cancelled_by_user = [False]
    prog       = ProgressForm()
    prog.show_with(cancelled_by_user)
    start_time = DateTime.Now

    items = par_form.items_out

    # Interference kontaklari + force okunan kontaklar (tekrarsiz) icin CID ata
    uniq = []
    keys = []
    for it in items:
        for c in (it["contact"], it["fcontact"]):
            k = ckey(c)
            if k not in keys:
                keys.append(k)
                uniq.append((c, get_existing_cid(c)))
    cid_list, next_cid, reused, created = assign_cids(uniq, start_cid, prog, cancelled_by_user)

    if cancelled_by_user[0]:
        prog.Close()
        MessageBox.Show(u"Operation cancelled by user.", u"Cancelled",
                        MessageBoxButtons.OK, MessageBoxIcon.Warning)
        return
    cmap = dict(zip(keys, cid_list))

    cs_map = dict(get_cs_list(model))
    for it in items:
        it["cid"]    = cmap[ckey(it["contact"])]
        it["fcid"]   = cmap[ckey(it["fcontact"])]
        it["cs_obj"] = cs_map.get(it["cs_name"])

    try:
        iters_done, err = run_iteration(static_analysis, items,
                                        par_form.max_iter, par_form.truncate,
                                        par_form.final_solve,
                                        prog, cancelled_by_user)
    except Exception as ex:
        iters_done = 0
        err = u"Unexpected error: " + unicode(str(ex))

    elapsed_sec = int((DateTime.Now - start_time).TotalSeconds)
    prog.Close()

    lines = []
    for n_i, it in enumerate(items):
        if not it.get("hist"):
            lines.append(str(n_i + 1) + u".  " + it["contact"].Name + u":  no result")
            continue
        status = u"OK" if it["converged"] else u"NOT within tolerance"
        txt = (str(n_i + 1) + u".  " + it["contact"].Name + u"  (interference, cid_" + str(it["cid"]) + u")\n" +
               u"   force read at: " + it["fcontact"].Name + u"  (step " + str(it["step"]) + u", " + it["dir"] + u")\n" +
               u"   interference " + fmt_val(it["x"]) + u" mm   |   target " + fmt_val(it["target"] / 1000.0) +
               u" kN (" + tol_text(it) + u")   |   iteration result " + fmt_val(it["F"] / 1000.0) + u" kN  [" + status + u"]")
        if it.get("F_final") is not None:
            txt += u"\n   final reaction force (full solve): " + fmt_val(it["F_final"] / 1000.0) + u" kN"
        lines.append(txt)

    head = u""
    if err is not None:
        head = u"STOPPED: " + err + u"\n\n"
    elif cancelled_by_user[0]:
        head = u"Cancelled by user.\n\n"

    rs_txt = u""
    if RS_WARN:
        rs_txt = u"NOTE           : " + u"; ".join(RS_WARN) + u"\n"
    if FINAL_STATE["state"]:
        rs_txt += u"Solution       : " + FINAL_STATE["state"] + u"\n"

    MessageBox.Show(
        head +
        u"Solves run     : " + str(iters_done) + u"   (max " + str(par_form.max_iter) + u" iterations per step)\n" +
        u"Duration       : " + format_duration(elapsed_sec) + u"\n" + rs_txt + u"\n" +
        u"\n\n".join(lines),
        u"Iteration Done", MessageBoxButtons.OK,
        MessageBoxIcon.Information if err is None else MessageBoxIcon.Warning)

model       = ExtAPI.DataModel.Project.Model
connections = model.Connections

contact_tree = build_contact_tree(connections)
total_count  = sum(len(v) for (_, v) in contact_tree)
all_contacts = [(c.Name, c) for (g_name, c_list) in contact_tree for c in c_list]

if total_count == 0:
    MessageBox.Show(u"No contacts found in the analysis.", u"Info",
                    MessageBoxButtons.OK, MessageBoxIcon.Information)
else:
    start_cid       = find_max_existing_cid(connections)
    static_analysis = find_static_structural(model)

    if static_analysis is None:
        MessageBox.Show(u"No Static Structural analysis found in the model.", u"Error",
                        MessageBoxButtons.OK, MessageBoxIcon.Error)
    else:
        step_count = get_step_count(static_analysis)
        cs_names   = [n for (n, c) in get_cs_list(model)]

        type_form   = TypeSelectForm(start_cid, step_count)
        type_result = type_form.ShowDialog()

        while type_result == DialogResult.OK:
            cmd_type  = type_form.selected_type
            cmd_value = type_form.entered_value
            step_num  = type_form.entered_step
            is_iter   = (cmd_type == u"interference" and type_form.preload_mode == u"iteration")

            title_val   = u"iteration" if is_iter else cmd_value
            select_form = ContactSelectForm(contact_tree, cmd_type, title_val, start_cid)

            back_to_type = False
            while True:
                select_result = select_form.ShowDialog()
                if select_result == DialogResult.Retry:
                    back_to_type = True
                    break
                if select_result != DialogResult.OK:
                    break
                if not is_iter:
                    run_manual(static_analysis, select_form, cmd_type, cmd_value, step_num, start_cid)
                    break

                par_form   = IterParamForm(select_form.selected_contacts, step_count, cs_names, all_contacts)
                par_result = par_form.ShowDialog()
                if par_result == DialogResult.Retry:
                    continue
                if par_result == DialogResult.OK:
                    if LocationDefinitionMethod is None:
                        MessageBox.Show(u"LocationDefinitionMethod enum could not be imported; "
                                        u"force reaction probes cannot be created.", u"Error",
                                        MessageBoxButtons.OK, MessageBoxIcon.Error)
                    else:
                        run_iteration_mode(model, static_analysis, select_form, par_form, start_cid)
                break

            if back_to_type:
                type_form   = TypeSelectForm(start_cid, step_count)
                type_result = type_form.ShowDialog()
                continue
            break
