def style_button(b, kind):
    """kind: primary (mavi dolu) / danger (kirmizi dolu) / secondary (beyaz, mavi yazi)"""
    b.FlatStyle = FlatStyle.Flat
    b.Font = FB
    if kind == u"primary" or kind == u"danger":
        b.FlatAppearance.BorderSize = 0
        b.BackColor = ERR if kind == u"danger" else ACCENT
        b.ForeColor = Color.White
    else:
        b.FlatAppearance.BorderSize = 1
        b.BackColor = Color.White
        b.ForeColor = ACCENT


def apply_theme(root):
    """Eski formlardaki (ProgressForm, IterParamForm) butonlari Neuber stiline cevirir."""
    for c in root.Controls:
        try:
            if isinstance(c, Button):
                if c.Text == u"Stop":
                    style_button(c, u"danger")
                elif c.Text in PRIMARY_TEXTS:
                    style_button(c, u"primary")
                else:
                    style_button(c, u"secondary")
            elif isinstance(c, GroupBox) or isinstance(c, Panel):
                apply_theme(c)
        except:
            pass

def make_icon():
    """32x32 cark ikonu (8 dis, ortasi delik) - Neuber ile ayni."""
    bmp = Bitmap(32, 32)
    g = Graphics.FromImage(bmp)
    g.SmoothingMode = SmoothingMode.AntiAlias
    g.Clear(Color.Transparent)
    N, R, r, t1, t2 = 8, 14.0, 10.5, 0.42 * math.pi / 8, 0.62 * math.pi / 8
    path = GraphicsPath(FillMode.Alternate)
    pts = []
    for i in range(N):
        a = 2 * math.pi * i / N
        for rad, ang in ((r, a - t2), (R, a - t1), (R, a + t1), (r, a + t2)):
            pts.append(PointF(16 + rad * math.cos(ang), 16 + rad * math.sin(ang)))
    path.AddPolygon(Array[PointF](pts))
    path.AddEllipse(11.5, 11.5, 9.0, 9.0)          # delik
    g.FillPath(SolidBrush(ACCENT), path)
    g.Dispose()
    return Icon.FromHandle(bmp.GetHicon())

def fit_dropdown(combo, font, max_items=12):
    """Acilir liste makul yukseklikte (en cok max_items satir) ve en uzun metne gore genislikte olsun."""
    try:
        combo.MaxDropDownItems = max_items
        combo.IntegralHeight   = True
        w = combo.Width
        for itm in combo.Items:
            tw = TextRenderer.MeasureText(unicode(itm), font).Width + 30
            if tw > w:
                w = tw
        combo.DropDownWidth = min(w, 700)
    except Exception:

def format_duration(elapsed_sec):
    if elapsed_sec < 60:
        return str(elapsed_sec) + u" sec"
    else:
        mins = elapsed_sec // 60
        secs = elapsed_sec - mins * 60
        return str(mins) + u" min " + str(secs) + u" sec"

def fmt_val(x):
    t = (u"%.8f" % x).rstrip(u"0").rstrip(u".")
    if t == u"" or t == u"-":
        t = u"0"
    return t

HIDDEN_SUPPRESSED = [0]
MAX_DEPTH = 6


def get_existing_cid(contact):
    for child in contact.Children:
        if "CommandSnippet" not in child.GetType().ToString():
            continue
        try:
            matches = re.findall(r"cid_(\d+)\s*=\s*cid", child.Input)
            if matches:
                return int(matches[0])
        except:
            pass
    return None


def is_contact(obj):
    try:
        return "ContactRegion" in obj.GetType().ToString()
    except Exception:
        return False


def get_children(obj):
    try:
        return list(obj.Children)
    except Exception:
        return []


def find_max_existing_cid(obj, depth=0):
    """Tum kontaklardaki (alt basliklar dahil) max cid numarasi."""
    m = 0
    for ch in get_children(obj):
        if is_contact(ch):
            cid = get_existing_cid(ch)
            if cid is not None and cid > m:
                m = cid
        elif depth < MAX_DEPTH:
            m = max(m, find_max_existing_cid(ch, depth + 1))
    return m


def is_suppressed(obj):
    """Agacta 'x' simgesiyle gorunen (suppress edilmis) nesneleri birkac yontemle tespit eder."""
    try:
        if bool(obj.Suppressed):
            return True
    except Exception:
        pass
    try:
        if "suppress" in str(obj.ObjectState).lower():
            return True
    except Exception:
        pass
    try:
        if str(obj.Suppressed).strip().lower() in ("true", "yes", "1"):
            return True
    except Exception:
        pass
    return False


def collect_contacts(obj, path, depth, sup, found, order):
    """Alt basliklari gezerek kontaklari toplar. Ayni kontak birden fazla yerde gorunurse
    (ust klasor tum kontaklari duz listeleyebilir) EN DERIN yol (kendi klasoru) gecerli olur."""
    for ch in get_children(obj):
        if is_contact(ch):
            k = ckey(ch)
            s = sup or is_suppressed(ch)
            rec = found.get(k)
            if rec is None:
                found[k] = [ch, path, s]
                order.append(k)
            elif len(path) > len(rec[1]):
                rec[1] = path
                rec[2] = s
        elif depth < MAX_DEPTH:
            try:
                nm = unicode(ch.Name)
            except Exception:
                nm = u"?"
            collect_contacts(ch, path + [nm], depth + 1, sup or is_suppressed(ch), found, order)


def parent_path(c, root_key):
    """Kontagin .Parent zinciri Connections'a ulasiyorsa klasor adlari (ustten alta); aksi halde None."""
    names = []
    o = c
    for i in range(MAX_DEPTH + 2):
        try:
            p = o.Parent
        except Exception:
            return None
        if p is None:
            return None
        try:
            pk = ckey(p)
        except Exception:
            return None
        if pk == root_key:
            names.reverse()
            return names
        try:
            names.append(unicode(p.Name))
        except Exception:
            return None
        o = p
    return None


def prune_groups(g):
    g[2] = [s for s in g[2] if prune_groups(s)]
    return bool(g[1] or g[2])


def build_contact_tree(connections):
    """Kok dugum doner: [u"Connections", [kok kontaklar], [gruplar]]; kok dugumun kendisi agacta gosterilmez.
    Gruplar kontagin gercek klasoruna gore kurulur (Connections > Contacts > klasorler ...).
    Suppress edilmis kontaklar (ve gruplar) alinmaz."""
    HIDDEN_SUPPRESSED[0] = 0
    found = {}
    order = []
    collect_contacts(connections, [], 1, False, found, order)
    try:
        root_key = ckey(connections)
    except Exception:
        root_key = None
    root = [u"Connections", [], []]
    for k in order:
        c, path, sup = found[k]
        pp = parent_path(c, root_key) if root_key is not None else None
        if pp is not None and len(pp) > len(path):
            path = pp
        if sup:
            HIDDEN_SUPPRESSED[0] += 1
            continue
        node = root
        for nm in path:
            nxt = None
            for sg in node[2]:
                if sg[0] == nm:
                    nxt = sg
                    break
            if nxt is None:
                nxt = [nm, [], []]
                node[2].append(nxt)
            node = nxt
        node[1].append(c)
    prune_groups(root)
    return root


def count_contacts(g):
    return len(g[1]) + sum(count_contacts(s) for s in g[2])


def flatten_contacts(g):
    out = list(g[1])
    for s in g[2]:
        out.extend(flatten_contacts(s))
    return out

    # Belirli bir cid satirini tanimak icin (cid_1 ile cid_10 karismasin diye sondaki virgul var)

# Mevcut ayni step'teki preload command'i bul
# ---------------------------------------------------------------
def find_existing_solution_cmd(analysis, step_num):
    prefix = u"Rmodif,"
    for child in analysis.Children:
        if "CommandSnippet" not in child.GetType().ToString():
            continue
        try:
            if child.Name.startswith(u"PRELOAD_ITER_STEP"):
                continue
        except:
            pass
        try:
            snum = None
            try:
                snum = int(child.StepNumber)
            except:
                pass
            try:
                snum = int(child.InternalObject.StepNumber)
            except:
                pass
            if snum != step_num:
                continue
            if prefix.upper() in child.Input.upper():
                return child
        except:
            pass
    return None


# ---------------------------------------------------------------
# Mevcut command'e ekle veya yeni olustur
# ---------------------------------------------------------------
def add_or_update_solution_cmd(analysis, cmd_value, cid_list, step_num):
    existing  = find_existing_solution_cmd(analysis, step_num)
    new_lines = [make_cmd_line(cid_n, cmd_value) for cid_n in cid_list]

    if existing is not None:
        current_lines = [l for l in existing.Input.split(u"\n") if l.strip()]
        for line in new_lines:
            if line not in current_lines:
                current_lines.append(line)
        existing.Input = u"\n".join(current_lines)
        return existing, False
    else:
        cmd       = analysis.AddCommandSnippet()
        cmd.Name  = u"APDL_INTERFERENCE_STEP" + str(step_num)
        cmd.Input = u"\n".join(new_lines)
        cmd.StepSelectionMode = SequenceSelectionType.ByNumber
        cmd.StepNumber        = step_num
        return cmd, True


# ---------------------------------------------------------------
# Iterasyon icin: belirli cid satirinin DEGERINI degistir
# (satir varsa yerinde guncelle, yoksa ekle, command yoksa olustur)
# ---------------------------------------------------------------
def set_cmd_value(analysis, step_num, cid_n, value_str):
    existing = find_existing_solution_cmd(analysis, step_num)
    if existing is None:
        add_or_update_solution_cmd(analysis, value_str, [cid_n], step_num)
        return
    key       = cmd_line_key(cid_n)
    new_line  = make_cmd_line(cid_n, value_str)
    lines     = [l.strip() for l in existing.Input.split(u"\n") if l.strip()]
    found     = False
    for i in range(len(lines)):
        if lines[i].upper().startswith(key):
            lines[i] = new_line
            found = True
    if not found:
        lines.append(new_line)
    existing.Input = u"\n".join(lines)


# ---------------------------------------------------------------
# CID atama (hem manuel hem iterasyon modu kullanir)
# ---------------------------------------------------------------
def assign_cids(selected, start_cid, prog, cancelled):
    next_cid = start_cid
    cid_list = []
    reused   = 0
    created  = 0
    total    = len(selected)
    for idx, (contact, existing_cid) in enumerate(selected):
        if cancelled[0]:
            break
        prog.update(u"Adding CID commands...  (" + str(idx + 1) + u"/" + str(total) + u")",
                    contact.Name)
        if existing_cid is not None:
            cid_list.append(existing_cid)
            reused += 1
        else:
            next_cid += 1
            cid_list.append(next_cid)
            cmd_c       = contact.AddCommandSnippet()
            cmd_c.Name  = u"Contact_Command_" + str(next_cid)
            cmd_c.Input = u"cid_" + str(next_cid) + u" = cid"
            created += 1
    return cid_list, next_cid, reused, created

            self._log_nodes.append(n)
        self.CenterToScreen()

    # Deneme basladiginda satir hemen yazilir (interference degeri), cozum bitince force eklenir
    def begin_trial(self, i, text):
        try:
            n  = self._log_nodes[i]
            ch = n.Nodes.Add(text)
            ch.ForeColor = MUTED
            self._pending[i] = ch
            n.Expand()
            ch.EnsureVisible()
        except:
            pass
        Application.DoEvents()

    def close_pending(self, text):
        # iptal / hata durumunda yarim kalan denemeleri isaretle
        for i in list(self._pending.keys()):
            try:
                self._pending[i].Text = self._pending[i].Text + text
            except:
                pass
        self._pending = {}

    def log_trial(self, i, text, ok=None):
        try:
            n = self._log_nodes[i]
            if i in self._pending:
                ch = self._pending.pop(i)
                ch.Text = text
            else:
                ch = n.Nodes.Add(text)
            ch.ForeColor = OK if ok is True else DARK
            n.ForeColor = OK if ok is True else ACCENT   # kontak basligi: tuttu -> yesil
            n.Expand()
            ch.EnsureVisible()
        except:
            pass
        Application.DoEvents()

        cur = settings.CurrentStepNumber
    except:
        pass

        settings.CurrentStepNumber = s
        out[s] = float(settings.StepEndTime.Value)
    if cur is not None:

            settings.CurrentStepNumber = cur

            pass
    return out

SNIP_PREFIX = u"PRELOAD_ITER_STEP"

def build_abort_block():
    # /EOF: giris dosyasini burada bitirir; kontrol step'inden sonrasi cozulmez 
    return [u"! --- preload iteration: stop the solver here", u"/EOF"]

def remove_stale_snippets(analysis):
    for ch in list(analysis.Children):
        try:
            if ch.Name.startswith(SNIP_PREFIX):
                ch.Delete()
        except:
            pass

def add_read_snippets(analysis, items, ctrl_max):
    # Kontrol step'inden sonraki step'in basinda cozucuyu durdur (abort dosyasi)
    cmd       = analysis.AddCommandSnippet()
    cmd.Name  = SNIP_PREFIX + str(ctrl_max + 1)
    cmd.Input = u"\n".join(build_abort_block())
    cmd.StepSelectionMode = SequenceSelectionType.ByNumber
    cmd.StepNumber        = ctrl_max + 1
    return [cmd]

def delete_snippets(snips):
    for c in snips:
        try:
            c.Delete()

def html_esc(t):
    return unicode(t).replace(u"&", u"&amp;").replace(u"<", u"&lt;").replace(u">", u"&gt;")

def rows_to_html(rows, oks):
    """Excel kullanilamazsa ayni gorunumde basit HTML tablo."""
    h = [u"<table style='border-collapse:collapse;font-family:Calibri,Arial;font-size:11pt'>"]
    for r, row in enumerate(rows):
        if r == 0:
            bg, wt = u"#D9D9D9", u"bold"
        elif oks[r - 1]:
            bg, wt = u"#A9D08E", u"normal"
        else:
            bg, wt = u"#FFFFFF", u"normal"
        h.append(u"<tr>")
        for v in row:
            h.append(u"<td style='border:1px solid #808080;padding:2px 8px;background-color:" + bg +
                     u";font-weight:" + wt + u"'>" + html_esc(v) + u"</td>")
        h.append(u"</tr>")
    h.append(u"</table>")
    return u"".join(h)

def inline_excel_html(html):
    """Excel'in panodaki HTML'inden tabloyu alir; <style> sinif tanimlarini satir ici stile cevirir."""
    m = re.search(r"<!--StartFragment-->(.*?)<!--EndFragment-->", html, re.S)
    if m:
        frag = m.group(1)
    else:
        i = html.lower().index("<table")
        frag = html[i:]
    rules = {}
    for sm in re.finditer(r"<style[^>]*>(.*?)</style>", html, re.S | re.I):
        for mm in re.finditer(r"\.([A-Za-z0-9_]+)\s*\{([^}]*)\}", sm.group(1)):
            decls = [d.strip() for d in mm.group(2).split(u";") if d.strip() and not d.strip().lower().startswith(u"mso-")]
            rules[mm.group(1)] = u";".join(decls)
    def fix_tag(mt):
        tag = mt.group(0)
        cm = re.search(r"class=['\"]?([A-Za-z0-9_]+)['\"]?", tag)
        if not cm:
            return tag
        decl = rules.get(cm.group(1), u"")
        tag = tag.replace(cm.group(0), u"", 1)
        sm2 = re.search(r"style=(['\"])(.*?)\1", tag, re.S)
        if sm2:
            q = sm2.group(1)
            tag = tag.replace(sm2.group(0), u"style=" + q + decl + u";" + sm2.group(2) + q, 1)
        else:
            tag = tag[:-1] + u" style='" + decl + u"'>"
        return tag
    frag = re.sub(r"<(?:td|th|tr|table)\b[^>]*>", fix_tag, frag, flags=re.I)
    frag = re.sub(r"<!\[[^>]*\]>", u"", frag)
    if u"<table" not in frag.lower():
        raise Exception(u"no table in clipboard html")
    return frag

def excel_tables(items):
    """
    Her kontak icin Excel'de tablo olusturur, kopyalar ve panodaki HTML'i alir.
    Excel dosyasi hic kaydedilmez; is bitince kapatilir (silinecek dosya kalmaz). {item idx: html}
    """
    t = System.Type.GetTypeFromProgID("Excel.Application")
    if t is None:
        raise Exception(u"Excel is not installed")
    xl = System.Activator.CreateInstance(t)
    wb = None
    out = {}
    try:
        xl.Visible = False
        xl.DisplayAlerts = False
        wb = xl.Workbooks.Add()
        ws = wb.Worksheets[1]
        for it in items:
            if not it["hist"]:
                continue
            rows, oks = trial_rows(it)
            ws.Cells.Clear()
            for r, row in enumerate(rows):
                for c, v in enumerate(row):
                    ws.Cells[r + 1, c + 1].Value2 = v
            n = len(rows)
            ws.Range("A1:H" + str(n)).Borders.LineStyle = 1
            ws.Range("A1:H1").Font.Bold = True
            ws.Range("A1:H1").Interior.Color = 14277081
            for r, ok in enumerate(oks):
                if ok:
                    ws.Range("A" + str(r + 2) + ":H" + str(r + 2)).Interior.Color = 9359529
            ws.Columns("A:H").AutoFit()
            ws.Range("A1:H" + str(n)).Copy()
            html = read_clipboard_html()
            if html:
                try:
                    out[it["idx"]] = inline_excel_html(html)
                except Exception:
                    pass
    finally:
        try:
            xl.CutCopyMode = False
        except Exception:
            pass
        try:
            if wb is not None:
                wb.Close(False)
        except Exception:

        try:
            xl.Quit()
        except Exception:
            pass
        for o in (wb, xl):
            try:
                if o is not None:
                    Marshal.ReleaseComObject(o)
            except Exception:
                pass
        try:
            Clipboard.Clear()
        except Exception:
            pass
    return out

def create_probe(solution, name, contact, cs_obj):
    """Ayni adli probe daha once olusturulmussa YENISI ACILMAZ: mevcut probe kullanilir, ayarlari guncellenir
    ve altindaki eski comment'ler silinir (yeni comment iterasyon bitince yazilir). Boylece eski bir
    sonucun altinda yanlis bir comment kalmaz."""
    fr = None
    for ch in list(solution.Children):
        try:
            if ch.Name != name:
                continue
            if fr is None and is_force_probe(ch):
                fr = ch
            elif not is_force_probe(ch):
                ch.Delete()
        except:
            pass
    if fr is not None:
        for cm in list(fr.Children):
            try:
                if "Comment" in cm.GetType().ToString():
                    cm.Delete()
            except:
                pass
    else:
        fr = solution.AddForceReaction()
        fr.Name = name
    fr.LocationMethod = LocationDefinitionMethod.ContactRegion
    fr.ContactRegionSelection = contact
    if cs_obj is not None:
        try:
            fr.Orientation = cs_obj
        except:
            pass
    return fr

def within_tol(F, target, tol_pct, mode=u"both"):
    """mode: both = +/-tol ; lower = [hedef*(1-tol), hedef] ; upper = [hedef, hedef*(1+tol)]"""
    d    = F - target
    band = abs(target) * tol_pct / 100.0
    eps  = 1e-9 * max(abs(target), 1.0)
    if mode == u"lower":
        return (-band - eps) <= d <= eps
    if mode == u"upper":
        return -eps <= d <= (band + eps)
    return abs(d) <= band + eps

def tol_ok(it, F):
    return within_tol(F, it["target"], it["tol"], it.get("tol_mode", u"both"))

def tol_text(it):
    m = it.get("tol_mode", u"both")
    if m == u"lower":
        return u"-" + fmt_val(it["tol"]) + u"%"
    if m == u"upper":
        return u"+" + fmt_val(it["tol"]) + u"%"
    return u"+/-" + fmt_val(it["tol"]) + u"%"

def tol_cell(it):
    m = it.get("tol_mode", u"both")
    if m == u"lower":
        return u"-" + fmt_val(it["tol"]) + u"%"
    if m == u"upper":
        return u"+" + fmt_val(it["tol"]) + u"%"
    return fmt_val(it["tol"]) + u"%"

def aim_force(it):
    """Iterasyonun hedefledigi kuvvet: tek tarafli toleransta kabul bandinin ortasi."""
    m = it.get("tol_mode", u"both")
    if m == u"lower":
        return it["target"] * (1.0 - it["tol"] / 200.0)
    if m == u"upper":
        return it["target"] * (1.0 + it["tol"] / 200.0)
    return it["target"]

def next_value(hist, target):
    # hist: [(x, F), ...]   son nokta = hist[-1]
    x1, f1 = hist[-1]
    prev = None
    for k in range(len(hist) - 2, -1, -1):
        if abs(hist[k][0] - x1) > 1e-12 * max(1.0, abs(x1)):
            prev = hist[k]
            break
    xn = None
    if prev is not None and abs(f1 - prev[1]) > 1e-12:
        # sekant: interpolasyon / ekstrapolasyon
        xn = x1 + (target - f1) * (x1 - prev[0]) / (f1 - prev[1])
    else:
        # oran-oranti
        if abs(f1) < 1e-9:
            xn = x1 * 2.0
        else:
            xn = x1 * target / f1
    if xn <= 0:
        xn = x1 * 0.5
    if xn > 5.0 * x1:
        xn = 5.0 * x1
    if xn < 0.1 * x1:
        xn = 0.1 * x1
    return xn

def rs_abort_dirs(workdir):
    dirs = []
    if workdir:

            try:

ASYNC_SOLVE = [True]      # asenkron cozum sorun cikarirsa otomatik False olur (bekleyen Solve(True)'ya doner)

def rs_out_path(workdir):
    return os.path.join(workdir, u"solve.out") if workdir else None

def rs_mtime(pth):
    try:
        if pth and os.path.exists(pth):
            return os.path.getmtime(pth)
    except Exception:
        pass
    return 0.0

def rs_out_finished(pth, m0):
    """solve.out m0'dan sonra guncellenmis ve bitis ozeti ('Elapsed Time (sec)') iceriyor mu?"""
    try:
        if rs_mtime(pth) <= m0:
            return False
        f = open(pth, "rb")
        try:
            f.seek(0, 2)
            size = f.tell()
            f.seek(max(0, size - 8000))
            tail = f.read()
        finally:
            f.close()
        return "Elapsed Time (sec)" in tail
    except Exception:
        return False

def rs_solver_errors(workdir):
    """solve.out'taki ilk hata satirlari (varsa) - hata mesajina eklenir."""
    pth = rs_out_path(workdir)
    if not pth:
        return u""
    try:
        f = open(pth, "r")
        try:
            lines = f.read().splitlines()
        finally:
            f.close()
    except Exception:
        return u""
    hits = [l.strip() for l in lines if ("*** ERROR ***" in l or "FATAL" in l.upper())]
    if not hits:
        return u""
    return u"  |  solver output: " + u" / ".join([unicode(h, "latin-1") for h in hits[:3]])

def solve_blocking(analysis):
    try:
        analysis.Solve(True)
        return None
    except Exception as ex:
        return unicode(str(ex))

def solve_interruptible(analysis, cancelled, workdir):
    """
    Cozumu ASENKRON baslatir (Solve(False)) ve beklerken arayuzu isler; boylece Stop
    cozum sirasinda da calisir: Stop'a basilirsa solver'a abort dosyasi (file.abt) yazilir.
    Bitis, SOLVE.OUT'un guncellenip bitis ozetini icermesi ile dogrulanir (durum bilgisi tek basina yetmez).
    Asenkron cozum calismazsa otomatik olarak bekleyen Solve(True)'ya doner.
    Cozucu hic calismadiysa HATA doner (sessizce 0 kuvvet okunmaz).
    Donen: (hata metni veya None, iptal edildi mi)
    """
    pth = rs_out_path(workdir)
    m0  = rs_mtime(pth)
    if (not ASYNC_SOLVE[0]) or (not rs_state_readable(analysis)) or pth is None:
        err = solve_blocking(analysis)
        if err is None and pth is not None and rs_mtime(pth) <= m0:
            err = u"The solver did not run (solve.out was not updated)."
        return err, False
    t0 = time.time()
    try:
        analysis.Solve(False)
    except Exception as ex:
        ASYNC_SOLVE[0] = False
        if u"async" not in RS_WARN:
            RS_WARN.append(u"async solve unavailable (" + unicode(str(ex))[:80] + u"): Stop works only between solves")
        err = solve_blocking(analysis)
        if err is None and rs_mtime(pth) <= m0:
            err = u"The solver did not run (solve.out was not updated)."
        return err, False
    seen_busy  = False
    abort_sent = False
    abort_t    = 0.0
    idle_since = None
    while True:
        Application.DoEvents()
        Thread.Sleep(200)
        busy = rs_is_solving(analysis)
        now  = time.time()
        if busy:
            seen_busy  = True
            idle_since = None
        elif idle_since is None:
            idle_since = now
        if cancelled[0] and not abort_sent:
            rs_send_abort(workdir)
            abort_sent = True
            abort_t = now
        fin = rs_out_finished(pth, m0)
        if (not busy) and fin:
            break
        # solve.out guncellenmiyorsa (cozucu baska klasorde calisiyor olabilir) durum bilgisine guven
        if (not busy) and seen_busy and idle_since is not None and (now - idle_since > 30) and rs_mtime(pth) <= m0:
            break
        if (not busy) and seen_busy and idle_since is not None and (now - idle_since > 30) and (now - rs_mtime(pth) > 30):
            break
        if (not seen_busy) and (not fin) and (now - t0 > 120):
            if abort_sent:
                break
            return u"Could not detect that the solver started (the solution never reported 'Solving').", False
        if abort_sent and (now - abort_t > 120):
            break
    if abort_sent:
        rs_clear_abort(workdir)
    if cancelled[0]:
        return None, True
    if rs_mtime(pth) <= m0:
        return u"The solver did not run (solve.out was not updated).", False
    return None, False

def run_iteration(analysis, items, max_iter, use_cut, final_solve, prog, cancelled):
    """Her iterasyon analizi BASTAN cozer. Analiz ayarlarina (restart dahil) hic dokunulmaz."""

    FINAL_STATE["state"] = u""

    finally:

            except Exception:

def _run_iteration_core(analysis, items, max_iter, use_cut, final_solve, prog, cancelled):
    """
    ASAMALI iterasyon: preload'lar step sirasina gore cozulur. Once en kucuk step'teki satirlar
    (kendi icinde birlikte) tutturulur, sonra bir sonraki step'e gecilir:  2 -> 3 -> 4 ...
    Her asamada cozum o step'te durdurulur (/EOF) ve her iterasyon analizi bastan cozer.
    """
    settings    = analysis.AnalysisSettings
    solution    = analysis.Solution
    total_steps = get_step_count(analysis)
    steps       = sorted(set([it["step"] for it in items]))
    ctrl_max    = steps[-1]
    last_cut    = bool(use_cut and ctrl_max < total_steps)
    err         = None
    workdir     = get_workdir(analysis)

    if use_cut and workdir is None and steps[0] < total_steps:
        return 0, u"Solver working directory could not be determined (analysis.WorkingDir)."

    remove_stale_snippets(analysis)

    prog.update(u"Preparing...", u"")
    end_times  = get_step_end_times(settings, steps)
    used_names = []
    for i, it in enumerate(items):
        it["idx"] = i
        try:
            it["cs_id"] = int(it["cs_obj"].CoordinateSystemID)
        except:
            it["cs_id"] = 0
        it["probe"]     = None
        it["end_time"]  = end_times[it["step"]]
        it["x"]         = it["x0"]
        it["written_x"] = None
        it["solved_x"]  = None
        it["hist"]      = []
        it["converged"] = False
        it["F"]         = None
        it["F_final"]   = None

    titles = []
    for n_i, it in enumerate(items):
        t = str(n_i + 1) + u".  " + it["contact"].Name
        if ckey(it["fcontact"]) != ckey(it["contact"]):
            t += u"   ->   force @ " + it["fcontact"].Name
        t += u"   [step " + str(it["step"]) + u",  target " + fmt_val(it["target"] / 1000.0) + u" kN " + tol_text(it) + u"]"
        titles.append(t)
    try:
        prog.enable_log(titles)
    except:
        pass

    iters_done = 0
    snips      = []
    try:
        for si, s in enumerate(steps):
            if err is not None or cancelled[0]:
                break
            stage_items = [it for it in items if it["step"] == s]
            cut_stage   = bool(use_cut and s < total_steps)

            # bu asamanin probe'lari (onceki asamalarinkiler yerinde kalir)
            for it in stage_items:
                pname = u"Preload_" + it["fcontact"].Name
                if pname in used_names:
                    pname = pname + u" #" + str(it["idx"] + 1)
                used_names.append(pname)
                it["probe"] = create_probe(solution, pname, it["fcontact"], it["cs_obj"])
            if cut_stage:
                snips = add_read_snippets(analysis, stage_items, s)

            for k in range(1, max_iter + 1):
                if cancelled[0]:
                    break
                iters_done += 1
                tag = (u"Step " + str(s) + u"  (" + str(si + 1) + u"/" + str(len(steps)) +
                       u")  -  iteration " + str(k) + u"/" + str(max_iter))

                # 1) bu asamadaki satirlarin komutlarini yaz
                prog.update(tag + u"  -  writing commands", u"")
                for it in stage_items:
                    if it["written_x"] is None or abs(it["written_x"] - it["x"]) > 0.0:
                        set_cmd_value(analysis, it["step"], it["cid"], fmt_val(it["x"]))
                        it["written_x"] = it["x"]

                # 1b) denemeyi hemen goster: interference degeri yazilir, force cozumden sonra eklenir
                for it in stage_items:
                    prog.begin_trial(it["idx"], u"Trial " + str(len(it["hist"]) + 1) + u":   preload " +
                                     fmt_val(it["x"]) + u" mm   ->   solving...")

                # 2) coz (her zaman bastan)
                if cut_stage:
                    prog.update(tag + u"  -  solving",
                                u"From the beginning  |  solver stops after step " + str(s) + u" of " + str(total_steps))
                    clean_iter_files(workdir, items)
                else:
                    prog.update(tag + u"  -  solving", u"From the beginning  |  full analysis")
                clear_generated(analysis)
                solve_err, was_canc = solve_interruptible(analysis, cancelled, workdir)
                if solve_err:
                    solve_err = u"Solve error: " + solve_err
                if cancelled[0] or was_canc:
                    break

                # 3) oku: TEK evaluate (durdurulan cozumde de tamamlanan step'ler icin)
                try:
                    if solve_err is None:
                        try:
                            stx = unicode(str(analysis.Solution.Status)).lower()
                        except Exception:
                            stx = u""
                        if cut_stage:
                            if u"fail" in stx:
                                solve_err = u"Solve failed (status: " + stx + u")"
                        elif not solve_ok(analysis):
                            solve_err = u"Solution did not finish successfully (status: " + stx + u")"
                    if solve_err is not None:
                        err = solve_err + rs_solver_errors(workdir)
                        break
                    prog.update(tag + u"  -  evaluating", u"")
                    fmap = read_probes_once(solution, stage_items)
                    for it in stage_items:
                        F = fmap[it["idx"]]
                        it["F"] = F
                        it["solved_x"] = it["x"]
                        it["hist"].append((it["x"], F))
                except Exception as ex2:
                    err = u"Could not read results: " + unicode(str(ex2))
                    break

                # 4) bu asamanin yakinsama kontrolu + yeni deger
                all_ok = True
                for it in stage_items:
                    it["converged"] = tol_ok(it, it["F"])
                    if not it["converged"]:
                        all_ok = False
                    prog.log_trial(it["idx"],
                                   u"Trial " + str(len(it["hist"])) + u":   preload " + fmt_val(it["x"]) + u" mm   ->   force " +
                                   fmt_val(it["F"] / 1000.0) + u" kN   [" + (u"OK" if it["converged"] else u"out of tolerance") + u"]",
                                   it["converged"])
                if all_ok or k == max_iter:
                    break
                for it in stage_items:
                    if not it["converged"]:
                        it["x"] = next_value(it["hist"], aim_force(it))

            # asama tutturulamadiysa DUR (sonraki asamalara gecme)
            if err is None and not cancelled[0]:
                bad = [it for it in stage_items if not it["converged"]]
                if bad:
                    err = (u"Step " + str(s) + u": the preload did not reach the target within " + str(max_iter) +
                           u" iterations. Not converged: " +
                           u", ".join([str(it["idx"] + 1) + u" (" + fmt_val(it["F"] / 1000.0) + u" kN, target " +
                                       fmt_val(it["target"] / 1000.0) + u" kN)" for it in bad]) +
                           u". The remaining steps were not processed; the last tried interference values are kept in the commands.")

            # asama bitti: bu asamanin durdurma snippet'ini kaldir
            delete_snippets(snips)
            snips = []
            clean_iter_files(workdir, items)
    finally:
        delete_snippets(snips)
        clean_iter_files(workdir, items)

    if cancelled[0]:
        return iters_done, None

    # 5) bitis: EN SON cozulen interference degerleri ile devam
    if err is None:
        for it in items:
            if len(it["hist"]) == 0:
                continue
            it["x"] = it["hist"][-1][0]
            it["F"] = it["hist"][-1][1]
            it["converged"] = tol_ok(it, it["F"])
            if it["written_x"] is None or abs(it["written_x"] - it["x"]) > 0.0:
                set_cmd_value(analysis, it["step"], it["cid"], fmt_val(it["x"]))
                it["written_x"] = it["x"]

        have_hist  = (len([it for it in items if len(it["hist"]) > 0]) == len(items))
        full_valid = False
        if last_cut:
            if final_solve and not cancelled[0] and have_hist:
                prog.update(u"Final solve...", u"From the beginning, all " + str(total_steps) + u" steps")
                try:
                    prog.update(u"Final solve...", u"Clearing the generated data, then solving from the beginning")
                    clear_generated(analysis)
                    serr, canc = solve_interruptible(analysis, cancelled, workdir)
                    if serr:
                        raise Exception(serr)
                    if canc or cancelled[0]:
                        return iters_done, None
                    full_valid = solve_ok(analysis)
                    if not full_valid:
                        err = u"Final solve did not finish successfully (status: " + unicode(str(analysis.Solution.Status)) + u")"
                except Exception as ex4:
                    err = u"Final solve error: " + unicode(str(ex4))
        else:
            full_valid = have_hist

        FINAL_STATE["valid"] = bool(full_valid)

        # Final reaction force'lar: TEK evaluate
        if full_valid:
            prog.update(u"Evaluating final reaction forces...", u"")
            try:
                fmap = read_probes_once(solution, items)
                for it in items:
                    it["F_final"] = fmap.get(it["idx"])
            except Exception:
                pass

        # Probe'lara bilgi amacli comment: Excel tablosu (kopyala -> comment'e yapistir), sonra Excel kapatilir
        if have_hist:
            prog.update(u"Creating the summary tables...", u"")
            try:
                htmls = build_comment_tables(items)
            except Exception:
                htmls = {}
            for it in items:
                try:
                    add_probe_comment(it, iters_done, htmls.get(it["idx"]))
                except:
                    pass

    return iters_done, err

class ContactTree(object):
    def __init__(self, parent, x, y, w, h, connections):
        self.connections  = connections
        self.start_cid    = 0
        self.all_contacts = []
        self._full        = []       # tum kontaklar (agac sirasinda)

        self._sel         = {}       # ckey -> True  (filtre degisse de secimler korunur)
        self._ordered     = []       # gorunen: (node, contact, existing_cid)
        self._groups      = []
        self._updating    = False

        gc = GroupBox()
        gc.Text     = u"Contacts  (check to select)"
        gc.Font     = FB
        gc.Location = Point(x, y)
        gc.Size     = Size(w, h)

        self.lblInfo = label(gc, u"", 12, 48, w - 24, 17, F, MUTED)
        self.lblInfo.Anchor = AnchorStyles.Top | AnchorStyles.Left | AnchorStyles.Right

        self.tv = TreeView()
        self.tv.Font          = F
        self.tv.Location      = Point(12, 70)
        self.tv.Size          = Size(w - 24, h - 134)
        self.tv.BorderStyle   = BorderStyle.FixedSingle
        self.tv.CheckBoxes    = True
        self.tv.ShowLines     = True
        self.tv.ShowPlusMinus = True
        self.tv.ShowRootLines = True
        self.tv.Anchor        = AnchorStyles.Top | AnchorStyles.Bottom | AnchorStyles.Left | AnchorStyles.Right
        gc.Controls.Add(self.tv)
        self.tv.AfterCheck += self._on_node_check

        self.lblCid = label(gc, u"", 12, h - 60, w - 24, 17, F, MUTED)
        self.lblCid.Anchor = AnchorStyles.Bottom | AnchorStyles.Left | AnchorStyles.Right

        b_all.Anchor = AnchorStyles.Bottom | AnchorStyles.Left

        b_clr.Anchor = AnchorStyles.Bottom | AnchorStyles.Left
        b_clr.Click += self._on_clear

        self.reload(False)

    # --- model / gorunum ---
    def reload(self, keep_checked=True):
        """Modeli yeniden okur (yeni CID'ler gorunsun diye); secimler ve filtre korunur."""
        if keep_checked:
            self._sync()
        else:
            self._sel = {}
        self.start_cid = find_max_existing_cid(self.connections)
        self._root = build_contact_tree(self.connections)
        self._full = list(flatten_contacts(self._root))
        self.all_contacts = [(c.Name, c) for c in self._full]
        self._populate()
        if self.start_cid > 0:
            self.lblCid.Text = (u"Existing CIDs up to cid_" + str(self.start_cid) +
                                u"  ->  new contacts will get cid_" + str(self.start_cid + 1) + u"+")
        else:
            self.lblCid.Text = u"No existing CIDs  ->  will start from cid_1"

    def _filter_text(self):
        try:
            return self.txtSearch.Text.strip().lower()
        except Exception:
            return u""

    def _filtered(self, g, f):
        """Filtreye uyan kontaklari ve bos olmayan alt gruplari tutan kopya."""
        cs  = [c for c in g[1] if f in c.Name.lower()]
        sgs = []
        for sg in g[2]:
            fs = self._filtered(sg, f)
            if count_contacts(fs) > 0:
                sgs.append(fs)
        return [g[0], cs, sgs]

    def _populate(self):
        f = self._filter_text()
        root = self._root if not f else self._filtered(self._root, f)
        self._updating = True
        self.tv.BeginUpdate()
        self.tv.Nodes.Clear()
        self._ordered = []
        self._groups  = []
        for c in root[1]:                      # gruba bagli olmayan kontaklar kokte
            self.tv.Nodes.Add(self._contact_node(c))
        for g in root[2]:
            self.tv.Nodes.Add(self._add_group(g))
        for (node, c, ex) in self._ordered:
            node.Checked = (ckey(c) in self._sel)
        self._refresh_groups()
        self._updating = False
        self.tv.ExpandAll()
        self.tv.EndUpdate()
        self._update_info()

    def _update_info(self):
        txt = (u"Showing " + str(len(self._ordered)) + u" of " + str(len(self._full)) +
               u"     |     selected: " + str(len(self._sel)) +
               u"     |     suppressed hidden: " + str(HIDDEN_SUPPRESSED[0]))
        self.lblInfo.Text = txt

    def _sync(self):
        """Gorunen dugumlerin isaretini _sel'e yazar (gizlenenlerin secimi korunur)."""
        for (node, c, ex) in self._ordered:
            k = ckey(c)
            if node.Checked:
                self._sel[k] = True
            elif k in self._sel:
                del self._sel[k]

    def _on_search(self, sender, e):
        self._sync()
        self._populate()

    def _refresh_groups(self):
        for gn in reversed(self._groups):          # once en ic gruplar
            gn.Checked = (gn.Nodes.Count > 0 and all(ch.Checked for ch in gn.Nodes))

    def checked_keys(self):
        self._sync()
        return [ckey(c) for c in self._full if ckey(c) in self._sel]

    def set_checked(self, keys):
        self._sel = {}
        for k in keys:
            self._sel[k] = True
        self._populate()

    def _contact_node(self, c):
        ex = get_existing_cid(c)
        if ex is not None:
            txt = c.Name + u"  [cid_" + str(ex) + u"]"
        else:
            txt = c.Name
        ch          = TreeNode(txt)
        ch.Tag      = u"CONTACT"
        ch.NodeFont = F
        self._ordered.append((ch, c, ex))
        return ch

    def _add_group(self, g):
        gn          = TreeNode(g[0] + u"  [" + str(count_contacts(g)) + u" contacts]")
        gn.Tag      = u"GROUP"
        gn.NodeFont = FB
        self._groups.append(gn)
        for c in g[1]:
            gn.Nodes.Add(self._contact_node(c))
        for sg in g[2]:
            gn.Nodes.Add(self._add_group(sg))
        return gn

    def selected(self):
        """Isaretli kontaklar (filtre ile gizlenenler dahil): [(contact, existing_cid)]"""
        self._sync()
        return [(c, get_existing_cid(c)) for c in self._full if ckey(c) in self._sel]

    def _set_desc(self, node, state):
        for child in node.Nodes:
            child.Checked = state
            self._set_desc(child, state)

    def _on_node_check(self, sender, e):
        if self._updating:
            return
        self._updating = True
        node = e.Node
        if str(node.Tag) == u"GROUP":
            self._set_desc(node, node.Checked)
        parent = node.Parent
        while parent is not None:
            parent.Checked = all(child.Checked for child in parent.Nodes)
            parent = parent.Parent
        self._updating = False
        self._sync()
        self._update_info()

    def _set_all(self, state):
        """Sadece ekranda gorunen (filtreden gecen) kontaklari etkiler."""
        self._updating = True
        for gn in self._groups:
            gn.Checked = state
        for (node, c, ex) in self._ordered:
            node.Checked = state
        self._updating = False
        self._sync()
        self._update_info()

    def _on_select_all(self, sender, e):
        self._set_all(True)

    def _on_clear(self, sender, e):
        self._set_all(False)

def prefs_file():
    try:
        base = System.Environment.GetFolderPath(System.Environment.SpecialFolder.ApplicationData)
        d = os.path.join(unicode(base), u"PreloadIterTool")
        if not os.path.isdir(d):
            os.makedirs(d)
        return os.path.join(d, u"contact_settings.json")
    except Exception:
        return None

def prefs_project_key(analysis):
    try:
        return unicode(analysis.WorkingDir).lower()
    except Exception:
        return u"default"

def pref_key(contact):
    return ckey(contact) + u"|" + unicode(contact.Name)

def _read_prefs_file():
    pth = prefs_file()
    if json is None or not pth or not os.path.exists(pth):
        return {}
    try:
        f = open(pth, "r")
        try:
            return json.load(f)
        finally:
            f.close()
    except Exception:
        return {}

def load_prefs(pkey):
    return _read_prefs_file().get(pkey, {})

def find_pref(prefs, contact):
    """Once tam anahtar (kimlik|isim); bulunamazsa ayni isimli TEK kayit."""
    if not prefs:
        return None
    k = pref_key(contact)
    if k in prefs:
        return prefs[k]
    nm = u"|" + unicode(contact.Name)
    cands = [v for (kk, v) in prefs.items() if kk.endswith(nm)]
    if len(cands) == 1:
        return cands[0]
    return None

def save_prefs(pkey, items):
    """Iterasyon bitse de bitmese de (hata/iptal dahil) son secimler ve son interference degeri kaydedilir."""
    pth = prefs_file()
    if json is None or not pth:
        return
    try:
        data = _read_prefs_file()
        proj = data.get(pkey, {})
        for it in items:
            try:
                mode = it.get("tol_mode", u"both")
                sign = u"-" if mode == u"lower" else (u"+" if mode == u"upper" else u"")
                x = it.get("x", it.get("x0"))
                proj[pref_key(it["contact"])] = {
                    "fkey":   ckey(it["fcontact"]),
                    "fname":  unicode(it["fcontact"].Name),
                    "step":   int(it["step"]),
                    "dir":    unicode(it["dir"]),
                    "cs":     it.get("cs_name"),
                    "target": float(it["target"]) / 1000.0,
                    "tol":    sign + fmt_val(it["tol"]),
                    "x":      float(x)}
            except Exception:
                pass
        data[pkey] = proj
        f = open(pth, "w")
        try:
            json.dump(data, f)
        finally:
            f.close()
    except Exception:
        pass


def snippet_step_number(ch):
    snum = None
    try:
        snum = int(ch.StepNumber)
    except Exception:
        pass
    try:
        snum = int(ch.InternalObject.StepNumber)
    except Exception:
        pass
    return snum

def find_existing_preloads(analysis):
    """
    Analizdeki command snippet'lerinde onceden yazilmis 'Rmodif,cid_N,10,deger' satirlari.
    Donen: {cid numarasi: (step, interference degeri)}  (ayni cid birden fazla step'te varsa en son step)
    """
    pat = re.compile(r"Rmodif\s*,\s*cid_(\d+)\s*,\s*10\s*,\s*([-+0-9.eE]+)", re.I)
    out = {}
    try:
        children = list(analysis.Children)
    except Exception:
        return out
    for ch in children:
        try:
            if "CommandSnippet" not in ch.GetType().ToString():
                continue
            st = snippet_step_number(ch)
            if st is None:
                continue
            for line in ch.Input.split(u"\n"):
                m = pat.search(line)
                if not m:
                    continue
                cid = int(m.group(1))
                try:
                    val = float(m.group(2))
                except Exception:
                    continue
                if cid not in out or st >= out[cid][0]:
                    out[cid] = (st, val)
        except Exception:
            pass
    return out

    def _on_start(self, sender, e):
        def num(tb):
            try:
                return float(tb.Text.strip().replace(u",", u"."))
            except Exception:
                return None
        out = []
        problems = []
        for i, (contact, ex_cid, cb_force, cb_step, cb_dir, cb_cs, t_target, t_tol, t_x0) in enumerate(self._rows):
            miss = []
            if cb_force.SelectedIndex < 0:
                miss.append(u"Read force at")
            if cb_step.SelectedIndex < 0:
                miss.append(u"Step")
            if cb_dir.SelectedIndex < 0:
                miss.append(u"Direction")
            if cb_cs.Items.Count > 0 and cb_cs.SelectedIndex < 0:
                miss.append(u"Coordinate system")
            target = num(t_target)
            tol_txt  = t_tol.Text.strip().replace(u",", u".")
            tol_mode = u"both"
            if tol_txt.startswith(u"-"):
                tol_mode = u"lower"
                tol_txt  = tol_txt[1:]
            elif tol_txt.startswith(u"+"):
                tol_mode = u"upper"
                tol_txt  = tol_txt[1:]
            try:
                tol = float(tol_txt)
            except Exception:
                tol = None
            x0     = num(t_x0)
            if target is None or target <= 0:
                miss.append(u"Target [kN]")
            if tol is None or tol <= 0:
                miss.append(u"Tolerance [%]")
            if x0 is None or x0 <= 0:
                miss.append(u"Initial [mm]")
            if miss:
                problems.append(str(i + 1) + u". " + contact.Name + u":  " + u", ".join(miss))
                continue
            cs_name = None
            if cb_cs.SelectedIndex >= 0:
                cs_name = unicode(cb_cs.SelectedItem)
            out.append({"contact": contact, "existing_cid": ex_cid,
                        "fcontact": self._all_contacts[cb_force.SelectedIndex][1],
                        "step": cb_step.SelectedIndex + 1,
                        "dir": IterParamForm.DIRS[cb_dir.SelectedIndex],
                        "cs_name": cs_name,
                        "target": target * 1000.0, "tol": tol, "tol_mode": tol_mode, "x0": x0})   # ekranda kN, icerde N
        mi = None
        try:
            mi = int(self.t_maxit.Text.strip())
        except Exception:
            mi = None
        if mi is None or mi < 1:
            problems.append(u"Max iterations (per step): enter a whole number >= 1")
        if problems:
            MessageBox.Show(u"Please fill in / correct the following:\n\n" + u"\n".join(problems),
                            u"Warning", MessageBoxButtons.OK, MessageBoxIcon.Warning)
            return
        self.items_out   = out
        self.max_iter    = mi
        self.truncate    = self.chk_trunc.Checked
        self.final_solve = self.chk_final.Checked
        self.DialogResult = DialogResult.OK
        self.Close()

    # --- MANUEL MOD ---
    def run_manual(self, sel, val, step_num):
        cancelled  = [False]
        prog       = ProgressForm()
        start_time = DateTime.Now
        err        = None
        res        = None
        self.Enabled = False
        try:
            prog.Owner = self                 # pencere kapaninca odak bu forma donsun
            prog.show_with(cancelled)
            start_cid = find_max_existing_cid(self.connections)
            cid_list, next_cid, reused, created = assign_cids(sel, start_cid, prog, cancelled)
            if not cancelled[0]:
                prog.update(u"Adding solution command to analysis...",
                            u"Step " + str(step_num) + u"  |  PRELOAD")
                sol_cmd, is_new = add_or_update_solution_cmd(self.analysis, val, cid_list, step_num)
                res = (start_cid, next_cid, reused, created, sol_cmd, is_new)
        except Exception as ex:
            err = unicode(str(ex))
        was_cancelled = cancelled[0]            # prog.Close() bu bayragi True yapar, once oku
        elapsed_sec   = int((DateTime.Now - start_time).TotalSeconds)
        self.Enabled = True                     # once ana form aktif olsun, sonra progress kapansin
        try:
            prog.Close()
        except Exception:
            pass
        self.Activate()

        if err is not None:
            self.status(u"Error: " + err, ERR)
        elif was_cancelled or res is None:
            self.status(u"Operation cancelled by user.", WARN)
        else:
            start_cid, next_cid, reused, created, sol_cmd, is_new = res
            self.status(u"Done.  Interference = " + val + u" mm  at step " + str(step_num) + u"  ->  " +
                        sol_cmd.Name + (u"  (new command)" if is_new else u"  (existing command updated)") +
                        u"\n" + str(len(sel)) + u" contact(s):  " + str(reused) + u" reused CID, " +
                        str(created) + u" new CID  |  " + format_duration(elapsed_sec), OK)
        try:
            self.ct.reload()                  # yeni CID'ler agacta gorunsun, secimler korunur
        except Exception:
            pass

def run_iteration_job(model, analysis, connections, par_form):
    """Pencere kapandiktan sonra calisir: Mechanical kilitlenmez. Doner: (durum_metni, renk), ozet_argumanlari"""
    items      = par_form.items_out
    cancelled  = [False]
    prog       = ProgressForm()
    start_time = DateTime.Now
    err        = None
    iters_done = 0
    aborted    = False            # CID atama sirasinda iptal edildi
    start_cid  = 0
    next_cid   = 0
    reused     = 0
    created    = 0
    try:
        prog.show_with(cancelled)
        start_cid = find_max_existing_cid(connections)

        # Interference kontaklari + force okunan kontaklar (tekrarsiz) icin CID ata
        uniq = []
        keys = []
        for it in items:
            for c in (it["contact"], it["fcontact"]):
                k = ckey(c)
                if k not in keys:
                    keys.append(k)
                    uniq.append((c, get_existing_cid(c)))
        cid_list, next_cid, reused, created = assign_cids(uniq, start_cid, prog, cancelled)
        if cancelled[0]:
            aborted = True
        else:
            cmap   = dict(zip(keys, cid_list))
            cs_map = dict(get_cs_list(model))
            for it in items:
                it["cid"]    = cmap[ckey(it["contact"])]
                it["fcid"]   = cmap[ckey(it["fcontact"])]
                it["cs_obj"] = cs_map.get(it["cs_name"])
            iters_done, err = run_iteration(analysis, items,
                                            par_form.max_iter, par_form.truncate,
                                            par_form.final_solve,
                                            prog, cancelled)
    except Exception as ex:
        iters_done = 0
        err = u"Unexpected error: " + unicode(str(ex))

    try:
        prog.close_pending(u"   (not completed)")
    except Exception:
        pass
    was_cancelled = cancelled[0]            # prog.Close() bu bayragi True yapar, once oku
    save_prefs(prefs_project_key(analysis), items)      # tamamlansa da tamamlanmasa da
    elapsed_sec = int((DateTime.Now - start_time).TotalSeconds)
    try:
        prog.Close()
    except Exception:
        pass

    if aborted:
        return (u"Operation cancelled by user.", WARN), None

    lines = []
    for n_i, it in enumerate(items):
        if not it.get("hist"):
            lines.append(str(n_i + 1) + u".  " + it["contact"].Name + u":  no result")
            continue
        st_txt = u"OK" if it["converged"] else u"NOT within tolerance"
        txt = (str(n_i + 1) + u".  " + it["contact"].Name + u"  (interference, cid_" + str(it["cid"]) + u")\n" +
               u"   force read at: " + it["fcontact"].Name + u"  (step " + str(it["step"]) + u", " + it["dir"] + u")\n" +
               u"   interference " + fmt_val(it["x"]) + u" mm   |   target " + fmt_val(it["target"] / 1000.0) +
               u" kN (" + tol_text(it) + u")   |   iteration result " + fmt_val(it["F"] / 1000.0) + u" kN  [" + st_txt + u"]")
        if it.get("F_final") is not None:
            txt += u"\n   final reaction force (full solve): " + fmt_val(it["F_final"] / 1000.0) + u" kN"
        lines.append(txt)

    body = u""
    if err is not None:
        body = u"STOPPED: " + err + u"\n\n"

    rs_txt = u""
    if RS_WARN:
        rs_txt = u"NOTE           : " + u"; ".join(RS_WARN) + u"\n"
    if FINAL_STATE["state"]:
        rs_txt += u"Solution       : " + FINAL_STATE["state"] + u"\n"

    body += (u"Solves run     : " + str(iters_done) + u"   (max " + str(par_form.max_iter) + u" iterations per step)\n" +
             u"Duration       : " + format_duration(elapsed_sec) + u"\n" + rs_txt + u"\n" +
             u"\n\n".join(lines))

    if err is not None:
        head, col = u"Iteration stopped", WARN
        short = err if len(err) <= 240 else err[:240] + u"..."
        stat = (u"Stopped: " + short, WARN)
    elif was_cancelled:
        head, col = u"Cancelled by user", WARN
        stat = (u"Cancelled by user.  Solves run: " + str(iters_done), WARN)
    else:
        head, col = u"Iteration finished", OK
        stat = (u"Iteration finished.  Solves run: " + str(iters_done) + u"  |  " +
                format_duration(elapsed_sec), OK)
    return stat, (u"Iteration Done", head, col, body)
