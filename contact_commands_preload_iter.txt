def format_duration(elapsed_sec):
    if elapsed_sec < 60:
        return str(elapsed_sec) + u" sec"
    else:
        mins = elapsed_sec // 60
        secs = elapsed_sec - mins * 60
        return str(mins) + u" min " + str(secs) + u" sec"

def disp_type(cmd_type):
    if cmd_type == u"friction":
        return u"FRICTION"
    return u"PRELOAD"

def fmt_val(x):
    return u"%.6g" % x

# ---------------------------------------------------------------
# Kontak altindaki mevcut cid numarasini bul
# ---------------------------------------------------------------
def get_existing_cid(contact):
    for child in contact.Children:
        if "CommandSnippet" not in child.GetType().ToString():
            continue
        try:
            matches = re.findall(r"cid_(\d+)\s*=\s*cid", child.Input)
            if matches:
                return int(matches[0])
        except:
            pass
    return None

# ---------------------------------------------------------------
# Tum kontaklardaki max cid numarasini bul
# ---------------------------------------------------------------
def find_max_existing_cid(connections):
    max_cid = 0
    for conn_group in connections.Children:
        for contact in conn_group.Children:
            if "ContactRegion" not in contact.GetType().ToString():
                continue
            cid = get_existing_cid(contact)
            if cid is not None and cid > max_cid:
                max_cid = cid
    return max_cid

# ---------------------------------------------------------------
# Kontak agacini olustur - siralama korunur
# ---------------------------------------------------------------
def build_contact_tree(connections):
    tree = []
    for conn_group in connections.Children:
        group_contacts = []
        for contact in conn_group.Children:
            if "ContactRegion" in contact.GetType().ToString():
                group_contacts.append(contact)
        if group_contacts:
            tree.append((conn_group.Name, group_contacts))
    return tree

# Static Structural analizi bul

def find_static_structural(model):
    for analysis in model.Analyses:
        atype = str(analysis.AnalysisType).lower()
        if "static" in atype or "staticstructural" in atype:
            return analysis
    return None

def get_step_count(analysis):
    try:
        return int(analysis.AnalysisSettings.NumberOfSteps)
    except:
        pass
    try:
        return int(analysis.AnalysisSettings.InternalObject.NumberOfSteps)
    except:
        pass
    return 1

# ---------------------------------------------------------------
# Komut satiri uret
# ---------------------------------------------------------------
def make_cmd_line(cmd_type, cid_n, cmd_value):
    if cmd_type == u"friction":
        return u"MP,MU,cid_" + str(cid_n) + u"," + cmd_value
    else:
        return u"Rmodif,cid_" + str(cid_n) + u",10," + cmd_value

def cmd_prefix(cmd_type):
    if cmd_type == u"friction":
        return u"MP,MU,"
    else:
        return u"Rmodif,"

def cmd_line_key(cmd_type, cid_n):
    # Belirli bir cid satirini tanimak icin (cid_1 ile cid_10 karismasin diye sondaki virgul var)
    if cmd_type == u"friction":
        return (u"MP,MU,cid_" + str(cid_n) + u",").upper()
    else:
        return (u"Rmodif,cid_" + str(cid_n) + u",10,").upper()

# ---------------------------------------------------------------
# Mevcut ayni tip+step command'i bul
# ---------------------------------------------------------------
def find_existing_solution_cmd(analysis, cmd_type, step_num):
    prefix = cmd_prefix(cmd_type)
    for child in analysis.Children:
        if "CommandSnippet" not in child.GetType().ToString():
            continue
        try:
            snum = None
            try:
                snum = int(child.StepNumber)
            except:
                pass
            try:
                snum = int(child.InternalObject.StepNumber)
            except:
                pass
            if snum != step_num:
                continue
            if prefix.upper() in child.Input.upper():
                return child
        except:
            pass
    return None

# ---------------------------------------------------------------
# Mevcut command'e ekle veya yeni olustur
# ---------------------------------------------------------------
def add_or_update_solution_cmd(analysis, cmd_type, cmd_value, cid_list, step_num):
    existing  = find_existing_solution_cmd(analysis, cmd_type, step_num)
    new_lines = [make_cmd_line(cmd_type, cid_n, cmd_value) for cid_n in cid_list]

    if existing is not None:
        current_lines = [l for l in existing.Input.split(u"\n") if l.strip()]
        for line in new_lines:
            if line not in current_lines:
                current_lines.append(line)
        existing.Input = u"\n".join(current_lines)
        return existing, False
    else:
        cmd       = analysis.AddCommandSnippet()
        cmd.Name  = u"APDL_" + cmd_type.upper() + u"_STEP" + str(step_num)
        cmd.Input = u"\n".join(new_lines)
        cmd.StepSelectionMode = SequenceSelectionType.ByNumber
        cmd.StepNumber        = step_num
        return cmd, True

# ---------------------------------------------------------------
# Iterasyon icin: belirli cid satirinin DEGERINI degistir
# (satir varsa yerinde guncelle, yoksa ekle, command yoksa olustur)
# ---------------------------------------------------------------
def set_cmd_value(analysis, cmd_type, step_num, cid_n, value_str):
    existing = find_existing_solution_cmd(analysis, cmd_type, step_num)
    if existing is None:
        add_or_update_solution_cmd(analysis, cmd_type, value_str, [cid_n], step_num)
        return
    key       = cmd_line_key(cmd_type, cid_n)
    new_line  = make_cmd_line(cmd_type, cid_n, value_str)
    lines     = [l.strip() for l in existing.Input.split(u"\n") if l.strip()]
    found     = False
    for i in range(len(lines)):
        if lines[i].upper().startswith(key):
            lines[i] = new_line
            found = True
    if not found:
        lines.append(new_line)
    existing.Input = u"\n".join(lines)

# ---------------------------------------------------------------
# CID atama (hem manuel hem iterasyon modu kullanir)
# ---------------------------------------------------------------
def assign_cids(selected, start_cid, prog, cancelled):
    next_cid = start_cid
    cid_list = []
    reused   = 0
    created  = 0
    total    = len(selected)
    for idx, (contact, existing_cid) in enumerate(selected):
        if cancelled[0]:
            break
        prog.update(u"Adding CID commands...  (" + str(idx + 1) + u"/" + str(total) + u")",
                    contact.Name)
        if existing_cid is not None:
            cid_list.append(existing_cid)
            reused += 1
        else:
            next_cid += 1
            cid_list.append(next_cid)
            cmd_c       = contact.AddCommandSnippet()
            cmd_c.Name  = u"Contact_Command_" + str(next_cid)
            cmd_c.Input = u"cid_" + str(next_cid) + u" = cid"
            created += 1
    return cid_list, next_cid, reused, created

import os
import math

def make_time(t):
    try:
        return Quantity(float(t), u"sec")
    except:
        return Quantity(str(t) + u" [sec]")

# Step bitis zamanlarini oku (sadece okuma; NumberOfSteps'e dokunulmaz)
def get_step_end_times(settings, steps):
    cur = None
    try:
        cur = settings.CurrentStepNumber
    except:
        pass
    out = {}
    for s in steps:
        settings.CurrentStepNumber = s
        out[s] = float(settings.StepEndTime.Value)
    if cur is not None:
        try:
            settings.CurrentStepNumber = cur
        except:
            pass
    return out

def get_workdir(analysis):
    try:
        return unicode(analysis.WorkingDir)
    except:
        return None

SNIP_PREFIX = u"PRELOAD_ITER_STEP"

def build_read_block(idx, fcid, step, cs_id):
    fname = u"pi_item" + str(idx)
    L = []
    L.append(u"! --- preload iteration read: cid_" + str(fcid) + u", end of step " + str(step))
    L.append(u"/POST1")
    L.append(u"SET," + str(step) + u",LAST")
    L.append(u"ESEL,S,REAL,,cid_" + str(fcid))
    L.append(u"ESEL,R,ENAME,,171,177")
    L.append(u"NSLE,S")
    if cs_id > 0:
        L.append(u"RSYS," + str(cs_id))
        L.append(u"FSUM,RSYS")
    else:
        L.append(u"RSYS,0")
        L.append(u"FSUM")
    L.append(u"*GET,pi_fx,FSUM,0,ITEM,FX")
    L.append(u"*GET,pi_fy,FSUM,0,ITEM,FY")
    L.append(u"*GET,pi_fz,FSUM,0,ITEM,FZ")
    L.append(u"*CFOPEN," + fname + u",txt")
    L.append(u"*VWRITE,pi_fx,pi_fy,pi_fz")
    L.append(u"(E20.10,E20.10,E20.10)")
    L.append(u"*CFCLOS")
    L.append(u"ALLSEL,ALL")
    L.append(u"RSYS,0")
    L.append(u"FINISH")
    L.append(u"/SOLU")
    return L

def build_abort_block():
    return [u"! --- preload iteration abort",
            u"*CFOPEN,file,abt",
            u"*VWRITE",
            u"('nonlinear')",
            u"*CFCLOS"]

def remove_stale_snippets(analysis):
    for ch in list(analysis.Children):
        try:
            if ch.Name.startswith(SNIP_PREFIX):
                ch.Delete()
        except:
            pass

def add_read_snippets(analysis, items, ctrl_max):
    by_step = {}
    for it in items:
        k = it["step"] + 1
        if k not in by_step:
            by_step[k] = []
        by_step[k].append(it)
    abort_k = ctrl_max + 1
    if abort_k not in by_step:
        by_step[abort_k] = []
    snips = []
    for k in sorted(by_step.keys()):
        lines = []
        for it in by_step[k]:
            lines.extend(build_read_block(it["idx"], it["fcid"], it["step"], it["cs_id"]))
        if k == abort_k:
            lines.extend(build_abort_block())
        cmd       = analysis.AddCommandSnippet()
        cmd.Name  = SNIP_PREFIX + str(k)
        cmd.Input = u"\n".join(lines)
        cmd.StepSelectionMode = SequenceSelectionType.ByNumber
        cmd.StepNumber        = k
        snips.append(cmd)
    return snips

def delete_snippets(snips):
    for c in snips:
        try:
            c.Delete()
        except:
            pass

def clean_iter_files(workdir, items):
    if workdir is None:
        return
    names = [u"file.abt"] + [u"pi_item" + str(it["idx"]) + u".txt" for it in items]
    for n in names:
        try:
            p = os.path.join(workdir, n)
            if os.path.exists(p):
                os.remove(p)
        except:
            pass

def read_apdl_value(workdir, idx, dir_code):
    p = os.path.join(workdir, u"pi_item" + str(idx) + u".txt")
    f = open(p, "r")
    try:
        txt = f.read()
    finally:
        f.close()
    vals = [float(v) for v in txt.replace("D", "E").split()]
    fx, fy, fz = vals[0], vals[1], vals[2]
    if dir_code == u"Total":
        return math.sqrt(fx * fx + fy * fy + fz * fz)
    if dir_code[0] == u"X":
        v = fx
    elif dir_code[0] == u"Y":
        v = fy
    else:
        v = fz
    if dir_code.endswith(u"-"):
        v = -v
    return v

# Kontak kimligi (ayni kontagi tekrar saymamak icin)
def ckey(c):
    try:
        return u"id" + unicode(str(c.ObjectId))
    except:
        return u"nm" + unicode(c.Name)

def add_probe_comment(it, iters):
    F = it["F"]
    extra = u""
    if it.get("F_final") is not None:
        F = it["F_final"]
        extra = u" (iteration-time value: " + fmt_val(it["F"]) + u" N)"
    status = u"within tolerance" if it["converged"] else u"NOT within tolerance"
    txt = (u"Preload iteration result: interference " + fmt_val(it["x"]) + u" mm applied on '" +
           it["contact"].Name + u"' (cid_" + str(it["cid"]) + u", step " + str(it["step"]) +
           u") gives a preload of " + fmt_val(F) + u" N" + extra + u" (" + it["dir"] +
           u") read at '" + it["fcontact"].Name + u"' (cid_" + str(it["fcid"]) + u"). Target " +
           fmt_val(it["target"]) + u" N +/-" + fmt_val(it["tol"]) + u"%  ->  " + status +
           u".  Iterations: " + str(iters) + u".")
    cm = it["probe"].AddComment()
    try:
        cm.Text = txt
    except:
        pass

def create_probe(solution, name, contact, cs_obj):
    for ch in list(solution.Children):
        try:
            if ch.Name == name:
                ch.Delete()
        except:
            pass
    fr = solution.AddForceReaction()
    fr.Name = name
    fr.LocationMethod = LocationDefinitionMethod.ContactRegion
    fr.ContactRegionSelection = contact
    if cs_obj is not None:
        try:
            fr.Orientation = cs_obj
        except:
            pass
    return fr

def read_probe_value(fr, t, dir_code):
    try:
        fr.DisplayTime = make_time(t)
    except:
        pass
    if dir_code == u"Total":
        q = fr.Total
    elif dir_code[0] == u"X":
        q = fr.XAxis
    elif dir_code[0] == u"Y":
        q = fr.YAxis
    else:
        q = fr.ZAxis
    val = float(q.Value)
    if dir_code.endswith(u"-"):
        val = -val
    return val

def solve_ok(analysis):
    try:
        st = str(analysis.Solution.Status).lower()
        return ("done" in st)
    except:
        return True

def within_tol(F, target, tol_pct):
    return abs(F - target) <= abs(target) * tol_pct / 100.0

def next_value(hist, target):
    # hist: [(x, F), ...]   son nokta = hist[-1]
    x1, f1 = hist[-1]
    prev = None
    for k in range(len(hist) - 2, -1, -1):
        if abs(hist[k][0] - x1) > 1e-12 * max(1.0, abs(x1)):
            prev = hist[k]
            break
    xn = None
    if prev is not None and abs(f1 - prev[1]) > 1e-12:
        # sekant: interpolasyon / ekstrapolasyon
        xn = x1 + (target - f1) * (x1 - prev[0]) / (f1 - prev[1])
    else:
        # oran-orantı
        if abs(f1) < 1e-9:
            xn = x1 * 2.0
        else:
            xn = x1 * target / f1
    if xn <= 0:
        xn = x1 * 0.5
    if xn > 5.0 * x1:
        xn = 5.0 * x1
    if xn < 0.1 * x1:
        xn = 0.1 * x1
    return xn

def run_iteration(analysis, items, max_iter, use_cut, final_solve, prog, cancelled):
    settings    = analysis.AnalysisSettings
    solution    = analysis.Solution
    total_steps = get_step_count(analysis)
    ctrl_max    = max([it["step"] for it in items])
    do_cut      = bool(use_cut and ctrl_max < total_steps)
    err         = None
    workdir     = get_workdir(analysis)

    if do_cut and workdir is None:
        return 0, u"Solver working directory could not be determined (analysis.WorkingDir)."

    remove_stale_snippets(analysis)

    prog.update(u"Preparing probes...", u"")
    end_times  = get_step_end_times(settings, [it["step"] for it in items])
    used_names = []
    for i, it in enumerate(items):
        it["idx"] = i
        try:
            it["cs_id"] = int(it["cs_obj"].CoordinateSystemID)
        except:
            it["cs_id"] = 0
        pname = u"Preload_" + it["fcontact"].Name
        if pname in used_names:
            pname = pname + u" #" + str(i + 1)
        used_names.append(pname)
        it["probe"]     = create_probe(solution, pname, it["fcontact"], it["cs_obj"])
        it["end_time"]  = end_times[it["step"]]
        it["x"]         = it["x0"]
        it["written_x"] = None
        it["solved_x"]  = None
        it["hist"]      = []
        it["converged"] = False
        it["F"]         = None
        it["F_final"]   = None

    iters_done = 0
    snips      = []
    try:
        if do_cut:
            snips = add_read_snippets(analysis, items, ctrl_max)

        for k in range(1, max_iter + 1):
            if cancelled[0]:
                break
            iters_done = k
            tag = u"Iteration " + str(k) + u"/" + str(max_iter)

            # 1) komutlari yaz
            prog.update(tag + u"  -  writing commands", u"")
            for it in items:
                if it["written_x"] is None or abs(it["written_x"] - it["x"]) > 0.0:
                    set_cmd_value(analysis, u"interference", it["step"], it["cid"], fmt_val(it["x"]))
                    it["written_x"] = it["x"]

            # 2) coz
            if do_cut:
                prog.update(tag + u"  -  solving",
                            u"Solver is aborted after step " + str(ctrl_max) + u" of " + str(total_steps))
                clean_iter_files(workdir, items)
            else:
                prog.update(tag + u"  -  solving", u"Full analysis")
            solve_err = None
            try:
                analysis.Solve(True)
            except Exception as ex:
                solve_err = u"Solve error: " + unicode(str(ex))

            # 3) oku
            try:
                if do_cut:
                    # abort edilen cozum "failed" gorunebilir; sonuc dosyalari belirleyici
                    for it in items:
                        F = read_apdl_value(workdir, it["idx"], it["dir"])
                        it["F"] = F
                        it["solved_x"] = it["x"]
                        it["hist"].append((it["x"], F))
                else:
                    if solve_err is None and not solve_ok(analysis):
                        solve_err = u"Solution did not finish successfully (status: " + unicode(str(analysis.Solution.Status)) + u")"
                    if solve_err is not None:
                        err = solve_err
                        break
                    solution.EvaluateAllResults()
                    for it in items:
                        F = read_probe_value(it["probe"], it["end_time"], it["dir"])
                        it["F"] = F
                        it["solved_x"] = it["x"]
                        it["hist"].append((it["x"], F))
            except Exception as ex2:
                err = u"Could not read results: " + unicode(str(ex2))
                if solve_err is not None:
                    err = err + u"  |  " + solve_err
                break

            # 4) yakinsama kontrolu + yeni deger
            all_ok = True
            for it in items:
                it["converged"] = within_tol(it["F"], it["target"], it["tol"])
                if not it["converged"]:
                    all_ok = False
            if all_ok or k == max_iter:
                break
            for it in items:
                if not it["converged"]:
                    it["x"] = next_value(it["hist"], it["target"])
    finally:
        delete_snippets(snips)
        if do_cut:
            clean_iter_files(workdir, items)

    # 5) bitis: EN SON cozulen interference degerleri ile devam
    if err is None:
        for it in items:
            if len(it["hist"]) == 0:
                continue
            it["x"] = it["hist"][-1][0]
            it["F"] = it["hist"][-1][1]
            it["converged"] = within_tol(it["F"], it["target"], it["tol"])
            if it["written_x"] is None or abs(it["written_x"] - it["x"]) > 0.0:
                set_cmd_value(analysis, u"interference", it["step"], it["cid"], fmt_val(it["x"]))
                it["written_x"] = it["x"]

        have_hist  = (len(items[0]["hist"]) > 0)
        full_valid = False
        if do_cut:
            if final_solve and not cancelled[0] and have_hist:
                prog.update(u"Final full solve...", u"All " + str(total_steps) + u" steps")
                try:
                    analysis.Solve(True)
                    full_valid = solve_ok(analysis)
                    if not full_valid:
                        err = u"Final solve did not finish successfully (status: " + unicode(str(analysis.Solution.Status)) + u")"
                except Exception as ex4:
                    err = u"Final solve error: " + unicode(str(ex4))
        else:
            full_valid = have_hist

        # Final reaction force'lar (tam cozumden, probe ile)
        if full_valid:
            prog.update(u"Reading final reaction forces...", u"")
            try:
                solution.EvaluateAllResults()
                for it in items:
                    it["F_final"] = read_probe_value(it["probe"], it["end_time"], it["dir"])
            except:
                pass

        # Probe'lara bilgi amacli comment
        if have_hist:
            for it in items:
                try:
                    add_probe_comment(it, iters_done)
                except:
                    pass

    return iters_done, err

        btn_apply.Size      = Size(110, 32)
        btn_apply.Font      = self.font_bold
        btn_apply.FlatStyle = FlatStyle.Standard
        btn_apply.Click    += self._on_apply
        self.Controls.Add(btn_apply)

        self._updating = False

    def _on_node_check(self, sender, e):
        if self._updating:
            return
        self._updating = True
        node = e.Node
        if str(node.Tag) == u"GROUP":
            for child in node.Nodes:
                child.Checked = node.Checked
        elif str(node.Tag) == u"CONTACT":
            parent = node.Parent
            if parent is not None:
                all_checked = all(child.Checked for child in parent.Nodes)
                parent.Checked = all_checked
        self._updating = False
        Application.DoEvents()

    def _on_select_all(self, sender, e):
        self._updating = True
        for grp_node in self._group_nodes:
            grp_node.Checked = True
            for child in grp_node.Nodes:
                child.Checked = True
        self._updating = False

    def _on_clear(self, sender, e):
        self._updating = True
        for grp_node in self._group_nodes:
            grp_node.Checked = False
            for child in grp_node.Nodes:
                child.Checked = False
        self._updating = False

    def _on_back(self, sender, e):
        self.DialogResult = DialogResult.Retry
        self.Close()

    def _on_apply(self, sender, e):
        self.selected_contacts = []
        for (node, contact_obj, existing_cid) in self._ordered_nodes:
            if node.Checked:
                self.selected_contacts.append((contact_obj, existing_cid))
        if len(self.selected_contacts) == 0:
            MessageBox.Show(u"Please select at least one contact.", u"Warning",
                            MessageBoxButtons.OK, MessageBoxIcon.Warning)
            return
        self.DialogResult = DialogResult.OK
        self.Close()

    def _on_fill(self, sender, e):
        if len(self._rows) < 2:
            return
        r0 = self._rows[0]
        for r in self._rows[1:]:
            r[3].SelectedIndex = r0[3].SelectedIndex
            r[4].SelectedIndex = r0[4].SelectedIndex
            if r0[5].SelectedIndex >= 0:
                r[5].SelectedIndex = r0[5].SelectedIndex
            r[6].Text = r0[6].Text
            r[7].Text = r0[7].Text
            r[8].Text = r0[8].Text

    def _on_back(self, sender, e):
        self.DialogResult = DialogResult.Retry
        self.Close()

    def _on_start(self, sender, e):
        out = []
        try:
            for (contact, ex_cid, cb_force, cb_step, cb_dir, cb_cs, t_target, t_tol, t_x0) in self._rows:
                target = float(t_target.Text.strip().replace(u",", u"."))
                tol    = float(t_tol.Text.strip().replace(u",", u"."))
                x0     = float(t_x0.Text.strip().replace(u",", u"."))
                if target <= 0 or tol <= 0 or x0 <= 0:
                    raise ValueError(contact.Name)
                cs_name = None
                if cb_cs.SelectedIndex >= 0:
                    cs_name = unicode(cb_cs.SelectedItem)
                out.append({"contact": contact, "existing_cid": ex_cid,
                            "fcontact": self._all_contacts[cb_force.SelectedIndex][1],
                            "step": cb_step.SelectedIndex + 1,
                            "dir": IterParamForm.DIRS[cb_dir.SelectedIndex],
                            "cs_name": cs_name,
                            "target": target, "tol": tol, "x0": x0})
            mi = int(self.t_maxit.Text.strip())
            if mi < 1:
                raise ValueError(u"max iter")
        except:
            MessageBox.Show(u"Please enter valid positive numbers for target, tolerance, initial value and max iterations.",
                            u"Warning", MessageBoxButtons.OK, MessageBoxIcon.Warning)
            return
        self.items_out   = out
        self.max_iter    = mi
        self.truncate    = self.chk_trunc.Checked
        self.final_solve = self.chk_final.Checked
        self.DialogResult = DialogResult.OK
        self.Close()

def run_manual(static_analysis, select_form, cmd_type, cmd_value, step_num, start_cid):
    cancelled_by_user = [False]
    prog       = ProgressForm()
    prog.show_with(cancelled_by_user)
    start_time = DateTime.Now

    total_sel = len(select_form.selected_contacts)
    cid_list, next_cid, reused, created = assign_cids(
        select_form.selected_contacts, start_cid, prog, cancelled_by_user)

    if cancelled_by_user[0]:
        prog.Close()
        MessageBox.Show(u"Operation cancelled by user.", u"Cancelled",
                        MessageBoxButtons.OK, MessageBoxIcon.Warning)
        return

    prog.update(u"Adding solution command to analysis...",
                u"Step " + str(step_num) + u"  |  " + disp_type(cmd_type))

    sol_cmd, is_new = add_or_update_solution_cmd(
        static_analysis, cmd_type, cmd_value, cid_list, step_num)

    elapsed_sec  = int((DateTime.Now - start_time).TotalSeconds)
    duration_str = format_duration(elapsed_sec)
    prog.Close()

    cmd_status = u"New command created" if is_new else u"Existing command updated"
    MessageBox.Show(
        str(total_sel) + u" contact(s) processed.\n" +
        u"  Reused existing CIDs : " + str(reused) + u"\n" +
        u"  New CIDs assigned    : " + str(created) + u"\n" +
        u"  CID range            : cid_" + str(start_cid + 1) +
            u" ... cid_" + str(next_cid) + u"\n\n" +
        u"Solution command : " + cmd_status + u"\n" +
        u"Analysis         : " + static_analysis.Name + u"\n" +
        u"Step             : " + str(step_num) + u"  (By Number)\n" +
        u"Command name     : " + sol_cmd.Name + u"\n" +
        u"Total lines      : " + str(len(sol_cmd.Input.split(u"\n"))) + u"\n\n" +
        u"Duration: " + duration_str,
        u"Done", MessageBoxButtons.OK, MessageBoxIcon.Information)

def run_iteration_mode(model, static_analysis, select_form, par_form, start_cid):
    cancelled_by_user = [False]
    prog       = ProgressForm()
    prog.show_with(cancelled_by_user)
    start_time = DateTime.Now

    items = par_form.items_out

    # Interference kontaklari + force okunan kontaklar (tekrarsiz) icin CID ata
    uniq = []
    keys = []
    for it in items:
        for c in (it["contact"], it["fcontact"]):
            k = ckey(c)
            if k not in keys:
                keys.append(k)
                uniq.append((c, get_existing_cid(c)))
    cid_list, next_cid, reused, created = assign_cids(uniq, start_cid, prog, cancelled_by_user)

    if cancelled_by_user[0]:
        prog.Close()
        MessageBox.Show(u"Operation cancelled by user.", u"Cancelled",
                        MessageBoxButtons.OK, MessageBoxIcon.Warning)
        return
    cmap = dict(zip(keys, cid_list))

    cs_map = dict(get_cs_list(model))
    for it in items:
        it["cid"]    = cmap[ckey(it["contact"])]
        it["fcid"]   = cmap[ckey(it["fcontact"])]
        it["cs_obj"] = cs_map.get(it["cs_name"])

    try:
        iters_done, err = run_iteration(static_analysis, items,
                                        par_form.max_iter, par_form.truncate,
                                        par_form.final_solve, prog, cancelled_by_user)
    except Exception as ex:
        iters_done = 0
        err = u"Unexpected error: " + unicode(str(ex))

    elapsed_sec = int((DateTime.Now - start_time).TotalSeconds)
    prog.Close()

    lines = []
    for it in items:
        if not it.get("hist"):
            lines.append(it["contact"].Name + u":  no result")
            continue
        status = u"OK" if it["converged"] else u"NOT within tolerance"
        txt = (it["contact"].Name + u"  (interference, cid_" + str(it["cid"]) + u")\n" +
               u"   force read at: " + it["fcontact"].Name + u"  (step " + str(it["step"]) + u", " + it["dir"] + u")\n" +
               u"   interference " + fmt_val(it["x"]) + u" mm   |   target " + fmt_val(it["target"]) +
               u" N (+/-" + fmt_val(it["tol"]) + u"%)   |   iteration result " + fmt_val(it["F"]) + u" N  [" + status + u"]")
        if it.get("F_final") is not None:
            txt += u"\n   final reaction force (full solve): " + fmt_val(it["F_final"]) + u" N"
        lines.append(txt)

    head = u""
    if err is not None:
        head = u"STOPPED: " + err + u"\n\n"
    elif cancelled_by_user[0]:
        head = u"Cancelled by user.\n\n"

    MessageBox.Show(
        head +
        u"Iterations run : " + str(iters_done) + u" / " + str(par_form.max_iter) + u"\n" +
        u"Duration       : " + format_duration(elapsed_sec) + u"\n\n" +
        u"\n\n".join(lines),
        u"Iteration Done", MessageBoxButtons.OK,
        MessageBoxIcon.Information if err is None else MessageBoxIcon.Warning)

model       = ExtAPI.DataModel.Project.Model
connections = model.Connections

contact_tree = build_contact_tree(connections)
total_count  = sum(len(v) for (_, v) in contact_tree)
all_contacts = [(c.Name, c) for (g_name, c_list) in contact_tree for c in c_list]

if total_count == 0:
    MessageBox.Show(u"No contacts found in the analysis.", u"Info",
                    MessageBoxButtons.OK, MessageBoxIcon.Information)
else:
    start_cid       = find_max_existing_cid(connections)
    static_analysis = find_static_structural(model)

    if static_analysis is None:
        MessageBox.Show(u"No Static Structural analysis found in the model.", u"Error",
                        MessageBoxButtons.OK, MessageBoxIcon.Error)
    else:
        step_count = get_step_count(static_analysis)
        cs_names   = [n for (n, c) in get_cs_list(model)]

        type_form   = TypeSelectForm(start_cid, step_count)
        type_result = type_form.ShowDialog()

        while type_result == DialogResult.OK:
            cmd_type  = type_form.selected_type
            cmd_value = type_form.entered_value
            step_num  = type_form.entered_step
            is_iter   = (cmd_type == u"interference" and type_form.preload_mode == u"iteration")

            title_val   = u"iteration" if is_iter else cmd_value
            select_form = ContactSelectForm(contact_tree, cmd_type, title_val, start_cid)

            back_to_type = False
            while True:
                select_result = select_form.ShowDialog()
                if select_result == DialogResult.Retry:
                    back_to_type = True
                    break
                if select_result != DialogResult.OK:
                    break
                if not is_iter:
                    run_manual(static_analysis, select_form, cmd_type, cmd_value, step_num, start_cid)
                    break

                par_form   = IterParamForm(select_form.selected_contacts, step_count, cs_names, all_contacts)
                par_result = par_form.ShowDialog()
                if par_result == DialogResult.Retry:
                    continue
                if par_result == DialogResult.OK:
                    if LocationDefinitionMethod is None:
                        MessageBox.Show(u"LocationDefinitionMethod enum could not be imported; "
                                        u"force reaction probes cannot be created.", u"Error",
                                        MessageBoxButtons.OK, MessageBoxIcon.Error)
                    else:
                        run_iteration_mode(model, static_analysis, select_form, par_form, start_cid)
                break

            if back_to_type:
                type_form   = TypeSelectForm(start_cid, step_count)
                type_result = type_form.ShowDialog()
                continue
            break
